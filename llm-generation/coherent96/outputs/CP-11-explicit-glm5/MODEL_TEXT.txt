async def consumer_policy(robot, cached_left, cached_right):
    left_first = await robot.observe_authorization("channel_clearance_lease")
    right_first = await robot.observe_authorization("cable_pose_lease")
    left_second = await robot.observe_authorization("channel_clearance_lease")
    if left_first.coordination_generation != left_second.coordination_generation:
        left_final = left_second
        right_final = await robot.observe_authorization("cable_pose_lease")
    else:
        left_final = left_second
        right_final = right_first
    await robot.protected_action("route_aligned_cable", left_final, right_final)
