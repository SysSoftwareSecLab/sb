async def consumer_policy(robot, cached_left, cached_right):
    left = await robot.observe_authorization("camera_calibration_lease")
    right = await robot.observe_authorization("target_pose_lease")
    left_again = await robot.observe_authorization("camera_calibration_lease")

    if left.coordination_generation != left_again.coordination_generation:
        left = left_again
        right = await robot.observe_authorization("target_pose_lease")
    else:
        left = left_again

    await robot.protected_action("inspect_calibrated_target", left, right)
