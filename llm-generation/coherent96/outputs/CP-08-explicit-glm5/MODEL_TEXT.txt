async def consumer_policy(robot, cached_left, cached_right):
    left1 = await robot.observe_authorization("tray_receiver_lease")
    right1 = await robot.observe_authorization("tray_custody_lease")
    left2 = await robot.observe_authorization("tray_receiver_lease")
    if left1.coordination_generation != left2.coordination_generation:
        right1 = await robot.observe_authorization("tray_custody_lease")
    await robot.protected_action("release_received_tray", left2, right1)
