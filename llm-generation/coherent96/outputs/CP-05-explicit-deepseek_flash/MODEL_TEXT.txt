async def consumer_policy(robot, cached_left, cached_right):
    left_receipt = await robot.observe_authorization("receiver_readiness_lease")
    right_receipt = await robot.observe_authorization("payload_custody_lease")
    left_again = await robot.observe_authorization("receiver_readiness_lease")

    if left_again.coordination_generation != left_receipt.coordination_generation:
        left_receipt = left_again
        right_receipt = await robot.observe_authorization("payload_custody_lease")
    else:
        left_receipt = left_again

    await robot.protected_action("release_supported_payload", left_receipt, right_receipt)
