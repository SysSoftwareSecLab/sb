async def consumer_policy(robot, cached_left, cached_right):
    left = await robot.observe_authorization("slot_clearance_lease")
    right = await robot.observe_authorization("tab_alignment_lease")
    left_again = await robot.observe_authorization("slot_clearance_lease")

    if left.coordination_generation != left_again.coordination_generation:
        left = left_again
        right = await robot.observe_authorization("tab_alignment_lease")
    else:
        left = left_again

    await robot.protected_action("seat_aligned_tab", left, right)
