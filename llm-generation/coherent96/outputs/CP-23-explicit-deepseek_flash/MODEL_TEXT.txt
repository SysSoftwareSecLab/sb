async def consumer_policy(robot, cached_left, cached_right):
    left1 = await robot.observe_authorization("pallet_support_lease")
    right1 = await robot.observe_authorization("case_custody_lease")
    left2 = await robot.observe_authorization("pallet_support_lease")

    if left1.coordination_generation != left2.coordination_generation:
        right2 = await robot.observe_authorization("case_custody_lease")
        await robot.protected_action("release_case_on_pallet", left2, right2)
    else:
        await robot.protected_action("release_case_on_pallet", left2, right1)
