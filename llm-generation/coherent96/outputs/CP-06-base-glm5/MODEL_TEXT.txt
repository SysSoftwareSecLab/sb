async def consumer_policy(robot, cached_left, cached_right):
    left_receipt = await robot.observe_authorization("human_zone_clearance")
    right_receipt = await robot.observe_authorization("part_custody_lease")
    if (
        left_receipt.state_id == "human_zone_clearance"
        and right_receipt.state_id == "part_custody_lease"
        and left_receipt.active
        and right_receipt.active
        and left_receipt.coordination_generation == right_receipt.coordination_generation
        and left_receipt.coordination_generation == left_receipt.version
        and right_receipt.coordination_generation == right_receipt.version
    ):
        await robot.protected_action("complete_guarded_handover", left_receipt, right_receipt)
