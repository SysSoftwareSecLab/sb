async def consumer_policy(robot, cached_left, cached_right):
    left1 = await robot.observe_authorization("slot_clearance_lease")
    right1 = await robot.observe_authorization("tab_alignment_lease")
    left2 = await robot.observe_authorization("slot_clearance_lease")
