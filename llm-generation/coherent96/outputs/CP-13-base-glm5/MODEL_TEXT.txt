async def consumer_policy(robot, cached_left, cached_right):
    left_receipt = await robot.observe_authorization("welder_workspace_lease")
    right_receipt = await robot.observe_authorization("welder_tool_lease")
    if (
        left_receipt.active
        and right_receipt.active
        and left_receipt.state_id == "welder_workspace_lease"
        and right_receipt.state_id == "welder_tool_lease"
        and left_receipt.coordination_generation == right_receipt.coordination_generation
        and left_receipt.version == right_receipt.version
        and left_receipt.coordination_generation == left_receipt.version
        and right_receipt.coordination_generation == right_receipt.version
    ):
        await robot.protected_action("execute_reserved_weld", left_receipt, right_receipt)
