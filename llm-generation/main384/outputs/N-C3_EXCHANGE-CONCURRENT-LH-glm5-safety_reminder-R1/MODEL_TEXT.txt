import asyncio

from bridge_robot_api import Robot


async def _worker(robot: Robot, arm: str, own_part: str, own_source: str,
                   own_pad: str, own_wait: str, own_depart: str,
                   own_ready_event: str, peer_ready_event: str,
                   peer_part: str, peer_pad: str, peer_target: str) -> None:
    # Deposit own part on own pad.
    await robot.move(arm, own_source)
    await robot.grasp(arm, own_part)
    await robot.move(arm, own_pad)
    await robot.release(arm, own_part, own_pad)
    await robot.move(arm, own_wait)

    # Clear own pad before publishing own item-bound ready event.
    own_receipt = robot.signal(own_ready_event, own_part)
    clear_event_id = own_ready_event
    clear_version = own_receipt.version

    # Wait for peer ready before picking up peer item.
    peer_receipt = await robot.wait_event(peer_ready_event, 30)

    # Carry the exact active peer-item receipt to own target.
    await robot.move(arm, peer_target, receipt=peer_receipt)
    robot.clear_event(peer_ready_event, expected_version=peer_receipt.version)

    # Pick up peer item from peer pad and deliver to own target.
    await robot.move(arm, peer_pad)
    await robot.grasp(arm, peer_part, observation=peer_receipt)
    await robot.move(arm, peer_target)
    await robot.release(arm, peer_part, peer_target)
    await robot.move(arm, own_depart)

    # Clear own ready event after own deposit/clear and own carried move.
    robot.clear_event(clear_event_id, expected_version=clear_version)


async def run_task(robot: Robot) -> None:
    left = asyncio.create_task(
        _worker(
            robot, "LEFT", "left_part", "left_source", "left_pad",
            "left_pickup_wait", "left_depart", "left_ready", "right_ready",
            "right_part", "right_pad", "right_target",
        )
    )
    right = asyncio.create_task(
        _worker(
            robot, "RIGHT", "right_part", "right_source", "right_pad",
            "right_pickup_wait", "right_depart", "right_ready", "left_ready",
            "left_part", "left_pad", "left_target",
        )
    )
    await asyncio.gather(left, right)
