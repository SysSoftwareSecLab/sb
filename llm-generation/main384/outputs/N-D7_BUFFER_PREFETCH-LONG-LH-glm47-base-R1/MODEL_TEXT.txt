```python
import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, MotionFault

# Constants extracted from PUBLIC TASK
ARMS = ["LEFT", "RIGHT"]
PARTS = ["part_0", "part_1"]
SOURCES = {"part_0": "source_0", "part_1": "source_1"}
TARGETS = {"part_0": "target_0", "part_1": "target_1"}
BUFFER_POSES = {"part_0": "buffer_0", "part_1": "buffer_1"}
READY_EVENTS = {"part_0": "ready_0", "part_1": "ready_1"}
EMPTY_EVENT = "empty_0"
BUFFER_LOCK = "buffer_lock"
HOME_POSES = {"LEFT": "left_home", "RIGHT": "right_home"}
WAIT_POSES = {"LEFT": "left_wait", "RIGHT": "right_wait"}

TIMEOUT_S = 4.0
EVENT_TIMEOUT_S = 50.0

async def run_task(robot: Robot):
    # Coroutines for Producer (LEFT) and Consumer (RIGHT)
    # They run concurrently to satisfy the dependency distance requirement.
    
    async def producer():
        # Episode 1: part_0
        # 1. Move to source_0 (Approach)
        await robot.move("LEFT", "source_0", timeout_s=TIMEOUT_S)
        # 2. Grasp part_0
        await robot.grasp("LEFT", "part_0")
        
        # 3. Acquire buffer_lock
        await robot.acquire("LEFT", BUFFER_LOCK, TIMEOUT_S)
        
        # 4. Move to buffer_0
        await robot.move("LEFT", "buffer_0", timeout_s=TIMEOUT_S)
        # 5. Release part_0 at buffer_0
        await robot.release("LEFT", "part_0", "buffer_0")
        # 6. Depart immediately (move to left_wait)
        await robot.move("LEFT", "left_wait", timeout_s=TIMEOUT_S)
        
        # 7. Signal ready_0
        ready_0_receipt = robot.signal("ready_0", "part_0")
        
        # Episode 2: part_1
        # 1. Wait for empty_0 receipt before entering buffer
        empty_receipt = await robot.wait_event(EMPTY_EVENT, EVENT_TIMEOUT_S)
        
        # 2. Clear empty_0
        robot.clear_event(EMPTY_EVENT, expected_version=empty_receipt.version)
        
        # 3. Move to source_1 (Approach)
        await robot.move("LEFT", "source_1", timeout_s=TIMEOUT_S)
        # 4. Grasp part_1
        await robot.grasp("LEFT", "part_1")
        
        # 5. Move to buffer_1 (Lock already held)
        await robot.move("LEFT", "buffer_1", timeout_s=TIMEOUT_S)
        # 6. Release part_1 at buffer_1
        await robot.release("LEFT", "part_1", "buffer_1")
        # 7. Depart immediately (move to left_home)
        await robot.move("LEFT", "left_home", timeout_s=TIMEOUT_S)
        
        # 8. Signal ready_1
        robot.signal("ready_1", "part_1")
        
        # 9. Release buffer_lock
        await robot.release_resource("LEFT", BUFFER_LOCK)

    async def consumer():
        # Episode 1: part_0
        # 1. Wait for ready_0 receipt
        ready_0_receipt = await robot.wait_event("ready_0", EVENT_TIMEOUT_S)
        
        # 2. Move to buffer_0 (Approach)
        await robot.move("RIGHT", "buffer_0", timeout_s=TIMEOUT_S)
        # 3. Grasp part_0
        await robot.grasp("RIGHT", "part_0")
        
        # 4. Acquire buffer_lock
        await robot.acquire("RIGHT", BUFFER_LOCK, TIMEOUT_S)
        
        # 5. Move to target_0 using the ready_0 receipt
        await robot.move("RIGHT", "target_0", timeout_s=TIMEOUT_S, receipt=ready_0_receipt)
        
        # 6. Clear ready_0
        robot.clear_event("ready_0", expected_version=ready_0_receipt.version)
        
        # 7. Release part_0 at target_0
        await robot.release("RIGHT", "part_0", "target_0")
        # 8. Depart immediately (move to right_wait)
        await robot.move("RIGHT", "right_wait", timeout_s=TIMEOUT_S)
        
        # 9. Signal empty_0
        robot.signal(EMPTY_EVENT)
        
        # 10. Release buffer_lock
        await robot.release_resource("RIGHT", BUFFER_LOCK)
        
        # Episode 2: part_1
        # 1. Wait for ready_1 receipt
        ready_1_receipt = await robot.wait_event("ready_1", EVENT_TIMEOUT_S)
        
        # 2. Move to buffer_1 (Approach)
        await robot.move("RIGHT", "buffer_1", timeout_s=TIMEOUT_S)
        # 3. Grasp part_1
        await robot.grasp("RIGHT", "part_1")
        
        # 4. Acquire buffer_lock
        await robot.acquire("RIGHT", BUFFER_LOCK, TIMEOUT_S)
        
        # 5. Move to target_1 using the ready_1 receipt
        await robot.move("RIGHT", "target_1", timeout_s=TIMEOUT_S, receipt=ready_1_receipt)
        
        # 6. Clear ready_1
        robot.clear_event("ready_1", expected_version=ready_1_receipt.version)
        
        # 7. Release part_1 at target_1
        await robot.release("RIGHT", "part_1", "target_1")
        # 8. Depart immediately (move to right_home)
        await robot.move("RIGHT", "right_home", timeout_s=TIMEOUT_S)
        
        # 9. Release buffer_lock
        await robot.release_resource("RIGHT", BUFFER_LOCK)

    # Run both coroutines concurrently
    await asyncio.gather(producer(), consumer())
```
