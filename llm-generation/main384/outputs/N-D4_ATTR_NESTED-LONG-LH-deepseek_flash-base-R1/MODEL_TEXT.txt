import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault


async def run_task(robot: Robot) -> None:
    # LONG variant: LEFT acquires tool before source pickup and keeps it across
    # the second-item wait and checks. Release tool on every normal exit.
    tool_acquired = False
    try:
        await robot.acquire("LEFT", "tool", 5.0)
        tool_acquired = True

        # ---------------- Episode 1: part_0 ----------------
        # Producer (LEFT): approach source_0 from left_home, grasp part_0
        await robot.move("LEFT", "left_home")
        await robot.move("LEFT", "source_0")
        await robot.grasp("LEFT", "part_0")

        # Acquire buffer_lock, enter buffer, place part_0, depart, release lock
        await robot.acquire("LEFT", "buffer_lock", 5.0)
        await robot.move("LEFT", "buffer_0")
        await robot.release("LEFT", "part_0", "buffer_0")
        await robot.move("LEFT", "left_wait")
        await robot.release_resource("LEFT", "buffer_lock")

        # Producer publishes ready_0
        robot.signal("ready_0", "part_0")

        # Consumer (RIGHT): wait ready_0, acquire lock, enter buffer, grasp part_0
        ready0 = await robot.wait_event("ready_0", 5.0)
        await robot.acquire("RIGHT", "buffer_lock", 5.0)
        await robot.move("RIGHT", "right_home")
        await robot.move("RIGHT", "buffer_0")
        await robot.grasp("RIGHT", "part_0")
        await robot.move("RIGHT", "right_wait")
        await robot.release_resource("RIGHT", "buffer_lock")

        # Consumer carries part_0 to target_0 with the exact active ready_0 receipt
        await robot.move("RIGHT", "target_0", receipt=ready0)
        await robot.release("RIGHT", "part_0", "target_0")
        await robot.move("RIGHT", "right_home")

        # Consumer clears ready_0 after carried move completes
        robot.clear_event("ready_0", expected_version=ready0.version)

        # Consumer publishes empty_0
        robot.signal("empty_0", "part_0")

        # ---------------- Episode 2: part_1 ----------------
        # Producer waits and clears empty_0 before entering buffer with part_1
        empty0 = await robot.wait_event("empty_0", 5.0)
        robot.clear_event("empty_0", expected_version=empty0.version)

        # Second-item checks: inspect line_clear (LEFT) and receiver_ready (RIGHT)
        # D4 joins them with asyncio.gather inside the i == 1 loop branch.
        obs_line, obs_recv = await asyncio.gather(
            robot.inspect("LEFT", "line_clear"),
            robot.inspect("RIGHT", "receiver_ready"),
        )
        line_val = obs_line.value
        recv_val = obs_recv.value
        if not (isinstance(line_val, dict) and line_val.get("clear") is True):
            raise ContractError("line_clear not clear")
        if not (isinstance(recv_val, dict) and recv_val.get("ready") is True):
            raise ContractError("receiver_ready not ready")

        # Producer (LEFT): approach source_1 from left_wait, grasp part_1
        await robot.move("LEFT", "left_wait")
        await robot.move("LEFT", "source_1")
        await robot.grasp("LEFT", "part_1")

        # Acquire buffer_lock, enter buffer, place part_1, depart, release lock
        await robot.acquire("LEFT", "buffer_lock", 5.0)
        await robot.move("LEFT", "buffer_1")
        await robot.release("LEFT", "part_1", "buffer_1")
        await robot.move("LEFT", "left_wait")
        await robot.release_resource("LEFT", "buffer_lock")

        # Producer publishes ready_1
        robot.signal("ready_1", "part_1")

        # Consumer (RIGHT): wait ready_1, acquire lock, enter buffer, grasp part_1
        ready1 = await robot.wait_event("ready_1", 5.0)
        await robot.acquire("RIGHT", "buffer_lock", 5.0)
        await robot.move("RIGHT", "right_wait")
        await robot.move("RIGHT", "buffer_1")
        await robot.grasp("RIGHT", "part_1")
        await robot.move("RIGHT", "right_wait")
        await robot.release_resource("RIGHT", "buffer_lock")

        # Consumer carries part_1 to target_1 with the exact active ready_1 receipt
        await robot.move("RIGHT", "target_1", receipt=ready1)
        await robot.release("RIGHT", "part_1", "target_1")
        await robot.move("RIGHT", "right_home")

        # Consumer clears ready_1 after carried move completes
        robot.clear_event("ready_1", expected_version=ready1.version)

        # Consumer publishes empty_0 again (buffer empty)
        robot.signal("empty_0", "part_1")

        # Return LEFT to home
        await robot.move("LEFT", "left_home")

    finally:
        if tool_acquired:
            try:
                await robot.release_resource("LEFT", "tool")
            except Exception:
                pass
