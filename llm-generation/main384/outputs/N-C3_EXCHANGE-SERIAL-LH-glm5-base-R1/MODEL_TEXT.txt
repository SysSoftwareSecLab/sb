import asyncio
from bridge_robot_api import Robot


async def _deposit_and_clear(robot: Robot, arm: str, part: str,
                             source_pose: str, pad_pose: str,
                             clear_pose: str, ready_event: str) -> None:
    """Deposit own part on own pad, clear pad, then publish item-bound ready."""
    await robot.move(arm, source_pose)
    await robot.grasp(arm, part)
    await robot.move(arm, pad_pose)
    await robot.release(arm, part, pad_pose)
    await robot.move(arm, clear_pose)
    robot.signal(ready_event, part)


async def _consume_and_depart(robot: Robot, arm: str, peer_part: str,
                              wait_pose: str, pad_pose: str,
                              target_pose: str, depart_pose: str,
                              ready_event: str) -> None:
    """Wait for peer ready, pick up peer item, carry to target, clear event, depart."""
    receipt = await robot.wait_event(ready_event, 30)
    await robot.move(arm, wait_pose)
    await robot.grasp(arm, peer_part)
    await robot.move(arm, target_pose, receipt=receipt)
    clear_event = robot.clear_event
    clear_event(ready_event, expected_version=receipt.version)
    await robot.release(arm, peer_part, target_pose)
    await robot.move(arm, depart_pose)


async def run_task(robot: Robot) -> None:
    # Variant A: deposit LEFT then RIGHT, then consume LEFT then RIGHT.
    await _deposit_and_clear(robot, "LEFT", "left_part",
                             "left_source", "left_pad",
                             "left_clear", "left_ready")
    await _deposit_and_clear(robot, "RIGHT", "right_part",
                             "right_source", "right_pad",
                             "right_clear", "right_ready")
    await _consume_and_depart(robot, "LEFT", "right_part",
                              "left_pickup_wait", "right_pad",
                              "right_target", "left_depart",
                              "right_ready")
    await _consume_and_depart(robot, "RIGHT", "left_part",
                              "right_pickup_wait", "left_pad",
                              "left_target", "right_depart",
                              "left_ready")
