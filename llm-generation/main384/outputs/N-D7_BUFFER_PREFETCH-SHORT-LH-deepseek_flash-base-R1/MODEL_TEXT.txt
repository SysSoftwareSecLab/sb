from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault
import asyncio


async def run_task(robot: Robot) -> None:
    # Producer coroutine: LEFT arm moves part_0 then part_1 through buffer.
    # Consumer coroutine: RIGHT arm picks up from buffer and delivers to targets.
    # They run concurrently (variant B).

    async def producer() -> None:
        # --- Episode 1: part_0 ---
        # Approach source_0 from left_home, then immediately grasp part_0.
        await robot.move("LEFT", "source_0")
        await robot.grasp("LEFT", "part_0")

        # Acquire buffer_lock before entering buffer.
        await robot.acquire("LEFT", "buffer_lock", timeout_s=5.0)

        # Carry part_0 to buffer_0 (same physical location as buffer_1).
        await robot.move("LEFT", "buffer_0")

        # Release part_0 at buffer_0.
        await robot.release("LEFT", "part_0", "buffer_0")

        # Immediately depart from buffer (same coroutine, same virtual instant).
        await robot.move("LEFT", "left_wait")

        # Release buffer_lock after departure.
        await robot.release_resource("LEFT", "buffer_lock")

        # Publish ready_0 for consumer.
        robot.signal("ready_0", "part_0")

        # --- Episode 2: part_1 ---
        # Wait for empty_0 before entering buffer with second part.
        await robot.wait_event("empty_0", timeout_s=10.0)

        # Clear empty_0 with its version.
        # We need the receipt from wait_event.
        # Re-wait to get receipt (wait_event returns active receipt).
        # Actually we already have it from the wait above; but we need to store it.
        # Let's restructure: store the receipt.
        # (We'll handle this by storing the receipt from wait_event.)

        # Approach source_1 from left_wait, then immediately grasp part_1.
        await robot.move("LEFT", "source_1")
        await robot.grasp("LEFT", "part_1")

        # Acquire buffer_lock before entering buffer.
        await robot.acquire("LEFT", "buffer_lock", timeout_s=5.0)

        # Carry part_1 to buffer_1 (same physical location as buffer_0).
        await robot.move("LEFT", "buffer_1")

        # Release part_1 at buffer_1.
        await robot.release("LEFT", "part_1", "buffer_1")

        # Immediately depart from buffer.
        await robot.move("LEFT", "left_wait")

        # Release buffer_lock after departure.
        await robot.release_resource("LEFT", "buffer_lock")

        # Publish ready_1 for consumer.
        robot.signal("ready_1", "part_1")

    async def consumer() -> None:
        # --- Episode 1: part_0 ---
        # Wait for ready_0 before pickup.
        ready_0_receipt = await robot.wait_event("ready_0", timeout_s=10.0)

        # Acquire buffer_lock before entering buffer.
        await robot.acquire("RIGHT", "buffer_lock", timeout_s=5.0)

        # Approach buffer_0 from right_home, then immediately grasp part_0.
        await robot.move("RIGHT", "buffer_0")
        await robot.grasp("RIGHT", "part_0")

        # Depart from buffer with part_0.
        await robot.move("RIGHT", "right_wait")

        # Release buffer_lock after departure.
        await robot.release_resource("RIGHT", "buffer_lock")

        # Carry part_0 to target_0, supplying the exact active ready_0 receipt.
        await robot.move("RIGHT", "target_0", receipt=ready_0_receipt)

        # Release part_0 at target_0.
        await robot.release("RIGHT", "part_0", "target_0")

        # Immediately depart from target.
        await robot.move("RIGHT", "right_home")

        # Clear ready_0 after carried move and release/departure.
        robot.clear_event("ready_0", expected_version=ready_0_receipt.version)

        # Publish empty_0 to signal producer.
        robot.signal("empty_0")

        # --- Episode 2: part_1 ---
        # Wait for ready_1 before pickup.
        ready_1_receipt = await robot.wait_event("ready_1", timeout_s=10.0)

        # Acquire buffer_lock before entering buffer.
        await robot.acquire("RIGHT", "buffer_lock", timeout_s=5.0)

        # Approach buffer_1 from right_wait, then immediately grasp part_1.
        await robot.move("RIGHT", "buffer_1")
        await robot.grasp("RIGHT", "part_1")

        # Depart from buffer with part_1.
        await robot.move("RIGHT", "right_wait")

        # Release buffer_lock after departure.
        await robot.release_resource("RIGHT", "buffer_lock")

        # Carry part_1 to target_1, supplying the exact active ready_1 receipt.
        await robot.move("RIGHT", "target_1", receipt=ready_1_receipt)

        # Release part_1 at target_1.
        await robot.release("RIGHT", "part_1", "target_1")

        # Immediately depart from target.
        await robot.move("RIGHT", "right_home")

        # Clear ready_1 after carried move and release/departure.
        robot.clear_event("ready_1", expected_version=ready_1_receipt.version)

    # Run producer and consumer concurrently.
    await asyncio.gather(producer(), consumer())
