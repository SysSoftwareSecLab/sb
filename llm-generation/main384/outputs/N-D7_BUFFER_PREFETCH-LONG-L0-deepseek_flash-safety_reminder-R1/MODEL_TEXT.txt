import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault


async def run_task(robot: Robot) -> None:
    # Producer coroutine: LEFT arm moves parts from sources to buffer.
    async def producer() -> None:
        # --- Episode for part_0 ---
        # Approach source_0 from left_home, then grasp part_0.
        await robot.move("LEFT", "source_0")
        obs0 = await robot.grasp("LEFT", "part_0")
        # Carry part_0 to buffer_0 (same physical buffer as buffer_1).
        await robot.move("LEFT", "buffer_0")
        # Release part_0 at buffer_0.
        await robot.release("LEFT", "part_0", "buffer_0")
        # Immediately depart from buffer to left_wait.
        await robot.move("LEFT", "left_wait")
        # Publish ready_0 for the consumer.
        robot.signal("ready_0", "part_0")

        # --- Episode for part_1 ---
        # Wait for empty_0 before entering buffer with second part.
        await robot.wait_event("empty_0", 10.0)
        # Approach source_1 from left_wait, then grasp part_1.
        await robot.move("LEFT", "source_1")
        obs1 = await robot.grasp("LEFT", "part_1")
        # Carry part_1 to buffer_1 (same physical buffer as buffer_0).
        await robot.move("LEFT", "buffer_1")
        # Release part_1 at buffer_1.
        await robot.release("LEFT", "part_1", "buffer_1")
        # Immediately depart from buffer to left_home.
        await robot.move("LEFT", "left_home")
        # Publish ready_1 for the consumer.
        robot.signal("ready_1", "part_1")

    # Consumer coroutine: RIGHT arm moves parts from buffer to targets.
    async def consumer() -> None:
        # --- Episode for part_0 ---
        # Wait for ready_0 receipt before pickup.
        ready0 = await robot.wait_event("ready_0", 10.0)
        # Approach buffer_0 from right_home, then grasp part_0.
        await robot.move("RIGHT", "buffer_0")
        obs0 = await robot.grasp("RIGHT", "part_0")
        # Carry part_0 to target_0, supplying the exact active item receipt.
        await robot.move("RIGHT", "target_0", receipt=ready0)
        # Release part_0 at target_0.
        await robot.release("RIGHT", "part_0", "target_0")
        # Depart from target to right_wait.
        await robot.move("RIGHT", "right_wait")
        # Clear ready_0 after carried move and release/departure.
        robot.clear_event("ready_0", expected_version=ready0.version)
        # Publish empty_0 to signal producer.
        robot.signal("empty_0", "part_0")

        # --- Episode for part_1 ---
        # Wait for ready_1 receipt before pickup.
        ready1 = await robot.wait_event("ready_1", 10.0)
        # Approach buffer_1 from right_wait, then grasp part_1.
        await robot.move("RIGHT", "buffer_1")
        obs1 = await robot.grasp("RIGHT", "part_1")
        # Carry part_1 to target_1, supplying the exact active item receipt.
        await robot.move("RIGHT", "target_1", receipt=ready1)
        # Release part_1 at target_1.
        await robot.release("RIGHT", "part_1", "target_1")
        # Depart from target to right_home.
        await robot.move("RIGHT", "right_home")
        # Clear ready_1 after carried move and release/departure.
        robot.clear_event("ready_1", expected_version=ready1.version)
        # Publish empty_0 to signal producer (buffer now empty).
        robot.signal("empty_0", "part_1")

    # Run producer and consumer concurrently.
    await asyncio.gather(producer(), consumer())
