import asyncio

async def run_task(robot):
    """
    Implements the D7_BUFFER_PREFETCH task for SHORT distance and LH layout.
    
    Strategy:
    - LEFT (Producer) and RIGHT (Consumer) run concurrently.
    - LEFT picks part_0, moves to buffer, releases, signals ready_0, departs.
    - RIGHT waits for ready_0, picks part_0, moves to target_0, releases, clears ready_0, signals empty_0, departs.
    - LEFT waits for empty_0, picks part_1, moves to buffer, releases, signals ready_1, departs.
    - RIGHT waits for ready_1, picks part_1, moves to target_1, releases, clears ready_1, departs.
    - Both return home.
    """
    
    # Helper to acquire buffer lock
    async def acquire_lock(arm):
        await robot.acquire(arm, "buffer_lock", 120.0)

    # Helper to release buffer lock
    async def release_lock(arm):
        await robot.release_resource(arm, "buffer_lock")

    # --- LEFT PRODUCER COROUTINE ---
    async def producer():
        # Episode 1: part_0
        # 1. Move to source_0 (Approach)
        await robot.move("LEFT", "source_0")
        # 2. Grasp part_0
        await robot.grasp("LEFT", "part_0")
        # 3. Acquire lock
        await acquire_lock("LEFT")
        # 4. Move to buffer_0
        await robot.move("LEFT", "buffer_0")
        # 5. Release part_0 at buffer_0
        await robot.release("LEFT", "part_0", "buffer_0")
        # 6. Signal ready_0
        robot.signal("ready_0")
        # 7. Depart buffer (move to left_wait)
        await robot.move("LEFT", "left_wait")
        # 8. Release lock
        await release_lock("LEFT")

        # Wait for empty_0 before proceeding with part_1
        # Task: "Producer waits and clears empty_0 before entering buffer with the second part."
        empty_0_receipt = await robot.wait_event("empty_0", 120.0)
        robot.clear_event("empty_0", expected_version=empty_0_receipt.version)

        # Episode 2: part_1
        # 1. Move to source_1 (Approach)
        await robot.move("LEFT", "source_1")
        # 2. Grasp part_1
        await robot.grasp("LEFT", "part_1")
        # 3. Acquire lock
        await acquire_lock("LEFT")
        # 4. Move to buffer_1
        await robot.move("LEFT", "buffer_1")
        # 5. Release part_1 at buffer_1
        await robot.release("LEFT", "part_1", "buffer_1")
        # 6. Signal ready_1
        robot.signal("ready_1")
        # 7. Depart buffer (move to left_home)
        await robot.move("LEFT", "left_home")
        # 8. Release lock
        await release_lock("LEFT")

    # --- RIGHT CONSUMER COROUTINE ---
    async def consumer():
        # Episode 1: part_0
        # 1. Wait for ready_0
        ready_0_receipt = await robot.wait_event("ready_0", 120.0)
        # 2. Move to buffer_0 (Approach)
        await robot.move("RIGHT", "buffer_0")
        # 3. Grasp part_0
        await robot.grasp("RIGHT", "part_0")
        # 4. Acquire lock
        await acquire_lock("RIGHT")
        # 5. Move to target_0 carrying part_0, supplying ready_0 receipt
        await robot.move("RIGHT", "target_0", receipt=ready_0_receipt)
        # 6. Release part_0 at target_0
        await robot.release("RIGHT", "part_0", "target_0")
        # 7. Clear ready_0
        robot.clear_event("ready_0", expected_version=ready_0_receipt.version)
        # 8. Depart target (move to right_wait)
        await robot.move("RIGHT", "right_wait")
        # 9. Release lock
        await release_lock("RIGHT")
        # 10. Signal empty_0
        robot.signal("empty_0")

        # Episode 2: part_1
        # 1. Wait for ready_1
        ready_1_receipt = await robot.wait_event("ready_1", 120.0)
        # 2. Move to buffer_1 (Approach)
        await robot.move("RIGHT", "buffer_1")
        # 3. Grasp part_1
        await robot.grasp("RIGHT", "part_1")
        # 4. Acquire lock
        await acquire_lock("RIGHT")
        # 5. Move to target_1 carrying part_1, supplying ready_1 receipt
        await robot.move("RIGHT", "target_1", receipt=ready_1_receipt)
        # 6. Release part_1 at target_1
        await robot.release("RIGHT", "part_1", "target_1")
        # 7. Clear ready_1
        robot.clear_event("ready_1", expected_version=ready_1_receipt.version)
        # 8. Depart target (move to right_home)
        await robot.move("RIGHT", "right_home")
        # 9. Release lock
        await release_lock("RIGHT")

    # Run both coroutines concurrently
    await asyncio.gather(producer(), consumer())
