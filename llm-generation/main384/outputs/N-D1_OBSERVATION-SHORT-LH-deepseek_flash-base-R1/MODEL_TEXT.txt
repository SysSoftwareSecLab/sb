from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault


async def run_task(robot: Robot) -> None:
    # ------------------------------------------------------------------
    # Phase 1: LEFT approaches left_source from left_home and grasps left_part
    # ------------------------------------------------------------------
    await robot.move("LEFT", "left_home")
    await robot.move("LEFT", "left_source")
    await robot.grasp("LEFT", "left_part")

    # ------------------------------------------------------------------
    # Phase 2: RIGHT approaches right_source from right_home and grasps right_part
    # ------------------------------------------------------------------
    await robot.move("RIGHT", "right_home")
    await robot.move("RIGHT", "right_source")
    await robot.grasp("RIGHT", "right_part")

    # ------------------------------------------------------------------
    # Phase 3: LEFT carries left_part to inspection and releases it there.
    # Release onto inspection invalidates quality (version bump).
    # Immediately after release, LEFT departs to left_wait.
    # ------------------------------------------------------------------
    await robot.move("LEFT", "inspection")
    await robot.release("LEFT", "left_part", "inspection")
    await robot.move("LEFT", "left_wait")

    # ------------------------------------------------------------------
    # Phase 4: RIGHT carries right_part to right_target and releases it.
    # ------------------------------------------------------------------
    await robot.move("RIGHT", "right_target")
    await robot.release("RIGHT", "right_part", "right_target")

    # ------------------------------------------------------------------
    # Phase 5: RIGHT inspects quality (pass 0). This is the newly issued
    # current quality record after the release onto inspection.
    # ------------------------------------------------------------------
    quality_obs = await robot.inspect("RIGHT", "quality")

    # ------------------------------------------------------------------
    # Phase 6: LEFT approaches inspection from left_wait and grasps
    # left_part using the current quality observation.
    # ------------------------------------------------------------------
    await robot.move("LEFT", "inspection")
    await robot.grasp("LEFT", "left_part", observation=quality_obs)

    # ------------------------------------------------------------------
    # Phase 7: Decide based on accept_by_pass[0].
    # The value is {"accept_by_pass": [true, true], "item_id": "left_part"}.
    # ------------------------------------------------------------------
    value = quality_obs.value
    accept_first = bool(value["accept_by_pass"][0])

    if accept_first:
        # Accept on first pass: carry left_part to left_target.
        await robot.move("LEFT", "left_target")
        await robot.release("LEFT", "left_part", "left_target")
    else:
        # Rework path: carry left_part to rework, release, then re-approach
        # from rework_wait and grasp again.
        await robot.move("LEFT", "rework")
        await robot.release("LEFT", "left_part", "rework")
        await robot.move("LEFT", "rework_wait")

        # Second inspection pass: RIGHT inspects quality again.
        quality_obs2 = await robot.inspect("RIGHT", "quality")

        # LEFT approaches rework from rework_wait and grasps left_part.
        await robot.move("LEFT", "rework")
        await robot.grasp("LEFT", "left_part", observation=quality_obs2)

        # Carry to left_target and release.
        await robot.move("LEFT", "left_target")
        await robot.release("LEFT", "left_part", "left_target")

    # ------------------------------------------------------------------
    # Phase 8: Both arms return to their departure poses, empty.
    # ------------------------------------------------------------------
    await robot.move("LEFT", "left_depart")
    await robot.move("RIGHT", "right_depart")
