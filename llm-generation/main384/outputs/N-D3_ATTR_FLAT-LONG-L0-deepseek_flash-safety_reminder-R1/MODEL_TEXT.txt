import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault


async def run_task(robot: Robot) -> None:
    # LONG variant: LEFT acquires tool before source pickup and keeps it across
    # the second-item wait and checks. Release tool on every normal, failure and
    # cancellation exit.

    tool_held = False
    buffer_lock_held = False

    async def release_tool():
        nonlocal tool_held
        if tool_held:
            try:
                await robot.release_resource("LEFT", "tool")
            except Exception:
                pass
            tool_held = False

    async def release_buffer_lock():
        nonlocal buffer_lock_held
        if buffer_lock_held:
            try:
                await robot.release_resource("LEFT", "buffer_lock")
            except Exception:
                pass
            buffer_lock_held = False

    async def producer():
        nonlocal tool_held, buffer_lock_held

        # Acquire tool before source pickup (LONG).
        await robot.acquire("LEFT", "tool", 5.0)
        tool_held = True

        # ---- Episode for part_0 ----
        # Approach source_0 from left_home, then grasp part_0.
        await robot.move("LEFT", "source_0")
        await robot.grasp("LEFT", "part_0")

        # Acquire buffer_lock, enter buffer, place part_0 at buffer_0.
        await robot.acquire("LEFT", "buffer_lock", 5.0)
        buffer_lock_held = True
        await robot.move("LEFT", "buffer_0")
        await robot.release("LEFT", "part_0", "buffer_0")
        # Immediately depart buffer.
        await robot.move("LEFT", "left_wait")
        await release_buffer_lock()

        # Publish ready_0 for part_0.
        robot.signal("ready_0", "part_0")

        # Wait for empty_0 before entering buffer with second part.
        receipt_empty = await robot.wait_event("empty_0", 30.0)
        robot.clear_event("empty_0", expected_version=receipt_empty.version)

        # ---- Episode for part_1 ----
        # Approach source_1 from left_wait, then grasp part_1.
        await robot.move("LEFT", "source_1")
        await robot.grasp("LEFT", "part_1")

        # Second-item checks: inspect line_clear with LEFT and receiver_ready
        # with RIGHT. D3 performs them serially.
        obs_line = await robot.inspect("LEFT", "line_clear")
        obs_recv = await robot.inspect("RIGHT", "receiver_ready")

        # Both public Boolean fields must permit transfer.
        line_val = obs_line.value
        recv_val = obs_recv.value
        if not (isinstance(line_val, dict) and line_val.get("clear") is True):
            raise ContractError("line_clear not clear")
        if not (isinstance(recv_val, dict) and recv_val.get("ready") is True):
            raise ContractError("receiver_ready not ready")

        # Acquire buffer_lock, enter buffer, place part_1 at buffer_1.
        await robot.acquire("LEFT", "buffer_lock", 5.0)
        buffer_lock_held = True
        await robot.move("LEFT", "buffer_1")
        await robot.release("LEFT", "part_1", "buffer_1")
        # Immediately depart buffer.
        await robot.move("LEFT", "left_wait")
        await release_buffer_lock()

        # Publish ready_1 for part_1.
        robot.signal("ready_1", "part_1")

        # Return to home.
        await robot.move("LEFT", "left_home")

        await release_tool()

    async def consumer():
        nonlocal buffer_lock_held

        # ---- Episode for part_0 ----
        # Wait for ready_0 receipt.
        receipt_ready0 = await robot.wait_event("ready_0", 30.0)

        # Acquire buffer_lock, enter buffer, pick up part_0.
        await robot.acquire("RIGHT", "buffer_lock", 5.0)
        buffer_lock_held = True
        await robot.move("RIGHT", "buffer_0")
        await robot.grasp("RIGHT", "part_0")
        # Depart buffer.
        await robot.move("RIGHT", "right_wait")
        await release_buffer_lock()

        # Carried move to target_0, supplying the exact active item receipt.
        await robot.move("RIGHT", "target_0", receipt=receipt_ready0)
        await robot.release("RIGHT", "part_0", "target_0")
        # Depart target.
        await robot.move("RIGHT", "right_wait")

        # Clear ready_0 after carried move, release, and departure.
        robot.clear_event("ready_0", expected_version=receipt_ready0.version)

        # Publish empty_0.
        robot.signal("empty_0")

        # ---- Episode for part_1 ----
        # Wait for ready_1 receipt.
        receipt_ready1 = await robot.wait_event("ready_1", 30.0)

        # Acquire buffer_lock, enter buffer, pick up part_1.
        await robot.acquire("RIGHT", "buffer_lock", 5.0)
        buffer_lock_held = True
        await robot.move("RIGHT", "buffer_1")
        await robot.grasp("RIGHT", "part_1")
        # Depart buffer.
        await robot.move("RIGHT", "right_wait")
        await release_buffer_lock()

        # Carried move to target_1, supplying the exact active item receipt.
        await robot.move("RIGHT", "target_1", receipt=receipt_ready1)
        await robot.release("RIGHT", "part_1", "target_1")
        # Depart target.
        await robot.move("RIGHT", "right_wait")

        # Clear ready_1 after carried move, release, and departure.
        robot.clear_event("ready_1", expected_version=receipt_ready1.version)

        # Return to home.
        await robot.move("RIGHT", "right_home")

    try:
        await asyncio.gather(producer(), consumer())
    except BaseException:
        await release_buffer_lock()
        await release_tool()
        raise
    finally:
        await release_buffer_lock()
        await release_tool()
