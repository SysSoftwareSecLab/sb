```python
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault
import asyncio


async def _inspect_allocation(robot: Robot, arm: str) -> Observation:
    return await robot.inspect(arm, "allocation")


async def _candidate(robot: Robot, arm: str) -> None:
    obs = await _inspect_allocation(robot, arm)
    value = obs.value
    selected = value["selected_arm"]
    if selected != arm:
        return
    await robot.move(arm, "left_home" if arm == "LEFT" else "right_home")
    await robot.move(arm, "shared_source")
    await robot.grasp(arm, "shared_part", observation=obs)
    await robot.move(arm, "shared_target")
    await robot.release(arm, "shared_part", "shared_target")
    await robot.move(arm, "left_depart" if arm == "LEFT" else "right_depart")


async def run_task(robot: Robot) -> None:
    # SERIAL structure: candidate LEFT then RIGHT serially.
    await _candidate(robot, "LEFT")
    await _candidate(robot, "RIGHT")
```
