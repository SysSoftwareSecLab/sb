import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, MotionFault

async def run_task(robot: Robot):
    # Constants
    LEFT = "LEFT"
    RIGHT = "RIGHT"
    PART_0 = "part_0"
    PART_1 = "part_1"
    
    # Poses
    LEFT_HOME = "left_home"
    LEFT_WAIT = "left_wait"
    RIGHT_HOME = "right_home"
    RIGHT_WAIT = "right_wait"
    
    SOURCE_0 = "source_0"
    SOURCE_1 = "source_1"
    BUFFER_0 = "buffer_0"
    BUFFER_1 = "buffer_1"
    TARGET_0 = "target_0"
    TARGET_1 = "target_1"
    
    # Resources
    BUFFER_LOCK = "buffer_lock"
    TOOL = "tool"
    
    # Events
    READY_0 = "ready_0"
    READY_1 = "ready_1"
    EMPTY_0 = "empty_0"
    
    # Facts
    FACT_LINE_CLEAR = "line_clear"
    FACT_RECEIVER_READY = "receiver_ready"
    
    # Helper to release tool safely
    async def release_tool_safely():
        try:
            await robot.set_mode(LEFT, TOOL, "OFF")
            await robot.release_resource(LEFT, TOOL)
        except Exception:
            pass

    # --- PART 0 ---
    
    # Producer (LEFT)
    # 1. Acquire buffer_lock
    await robot.acquire(LEFT, BUFFER_LOCK, 4.0)
    
    # 2. Acquire tool (SHORT: acquire before buffer placement)
    await robot.acquire(LEFT, TOOL, 4.0)
    await robot.set_mode(LEFT, TOOL, "LEFT_PROFILE")
    
    # 3. Move to source_0
    await robot.move(LEFT, SOURCE_0)
    
    # 4. Grasp part_0
    await robot.grasp(LEFT, PART_0)
    
    # 5. Move to buffer_0
    await robot.move(LEFT, BUFFER_0)
    
    # 6. Release part_0
    await robot.release(LEFT, PART_0, BUFFER_0)
    
    # 7. Depart buffer (Immediate departure)
    await robot.move(LEFT, LEFT_HOME)
    
    # 8. Release tool
    await robot.set_mode(LEFT, TOOL, "OFF")
    await robot.release_resource(LEFT, TOOL)
    
    # 9. Release buffer_lock
    await robot.release_resource(LEFT, BUFFER_LOCK)
    
    # 10. Signal ready_0
    ready_0_receipt = robot.signal(READY_0, PART_0)
    
    # Consumer (RIGHT)
    # 1. Wait for ready_0
    cons_ready_0 = await robot.wait_event(READY_0, 4.0)
    
    # 2. Acquire buffer_lock
    await robot.acquire(RIGHT, BUFFER_LOCK, 4.0)
    
    # 3. Move to buffer_0
    await robot.move(RIGHT, BUFFER_0)
    
    # 4. Grasp part_0
    await robot.grasp(RIGHT, PART_0)
    
    # 5. Depart buffer (Immediate departure)
    await robot.move(RIGHT, TARGET_0, receipt=cons_ready_0)
    
    # 6. Clear ready_0
    robot.clear_event(READY_0, expected_version=cons_ready_0.version)
    
    # 7. Release part_0
    await robot.release(RIGHT, PART_0, TARGET_0)
    
    # 8. Depart target
    await robot.move(RIGHT, RIGHT_HOME)
    
    # 9. Release buffer_lock
    await robot.release_resource(RIGHT, BUFFER_LOCK)
    
    # 10. Signal empty_0
    robot.signal(EMPTY_0)
    
    # --- PART 1 ---
    
    # Producer (LEFT)
    # 1. Wait and clear empty_0
    prod_empty_0 = await robot.wait_event(EMPTY_0, 4.0)
    robot.clear_event(EMPTY_0, expected_version=prod_empty_0.version)
    
    # 2. Acquire buffer_lock
    await robot.acquire(LEFT, BUFFER_LOCK, 4.0)
    
    # 3. Move to source_1
    await robot.move(LEFT, SOURCE_1)
    
    # 4. Grasp part_1
    await robot.grasp(LEFT, PART_1)
    
    # 5. Move to buffer_1
    await robot.move(LEFT, BUFFER_1)
    
    # 6. Release part_1
    await robot.release(LEFT, PART_1, BUFFER_1)
    
    # 7. Depart buffer
    await robot.move(LEFT, LEFT_WAIT)
    
    # 8. Release buffer_lock
    await robot.release_resource(LEFT, BUFFER_LOCK)
    
    # 9. Signal ready_1
    ready_1_receipt = robot.signal(READY_1, PART_1)
    
    # Consumer (RIGHT)
    # 1. Wait for ready_1
    cons_ready_1 = await robot.wait_event(READY_1, 4.0)
    
    # 2. Acquire buffer_lock
    await robot.acquire(RIGHT, BUFFER_LOCK, 4.0)
    
    # 3. Move to buffer_1
    await robot.move(RIGHT, BUFFER_1)
    
    # 4. Grasp part_1
    await robot.grasp(RIGHT, PART_1)
    
    # 5. Depart buffer
    await robot.move(RIGHT, RIGHT_WAIT)
    
    # 6. Release buffer_lock
    await robot.release_resource(RIGHT, BUFFER_LOCK)
    
    # 7. Perform checks (D4: joined concurrency)
    # SHORT: Acquire tool only after wait and checks
    # Acquire tool for LEFT
    await robot.acquire(LEFT, TOOL, 4.0)
    await robot.set_mode(LEFT, TOOL, "LEFT_PROFILE")
    
    # Run inspections
    obs_line_clear = robot.inspect(LEFT, FACT_LINE_CLEAR)
    obs_receiver_ready = robot.refresh(RIGHT, FACT_RECEIVER_READY)
    
    await asyncio.gather(obs_line_clear, obs_receiver_ready)
    
    # Verify checks
    if not obs_line_clear.result().value.get("clear"):
        raise ValueError("line_clear check failed")
    if not obs_receiver_ready.result().value.get("ready"):
        raise ValueError("receiver_ready check failed")
    
    # 8. Move to target_1 with receipt
    await robot.move(RIGHT, TARGET_1, receipt=cons_ready_1)
    
    # 9. Clear ready_1
    robot.clear_event(READY_1, expected_version=cons_ready_1.version)
    
    # 10. Release part_1
    await robot.release(RIGHT, PART_1, TARGET_1)
    
    # 11. Depart target
    await robot.move(RIGHT, RIGHT_HOME)
    
    # 12. Signal empty_0
    robot.signal(EMPTY_0)
    
    # Cleanup
    await release_tool_safely()
