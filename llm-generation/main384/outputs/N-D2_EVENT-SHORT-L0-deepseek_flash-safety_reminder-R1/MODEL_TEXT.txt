from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault
import asyncio


async def run_task(robot: Robot) -> None:
    # Phase 1: each arm deposits its own part on its own pad and immediately clears the pad.
    # LEFT: approach left_source from left_home, grasp left_part, move to left_pad, release, depart.
    await robot.move("LEFT", "left_source")
    await robot.grasp("LEFT", "left_part")
    await robot.move("LEFT", "left_pad")
    await robot.release("LEFT", "left_part", "left_pad")
    await robot.move("LEFT", "left_clear")

    # RIGHT: approach right_source from right_home, grasp right_part, move to right_pad, release, depart.
    await robot.move("RIGHT", "right_source")
    await robot.grasp("RIGHT", "right_part")
    await robot.move("RIGHT", "right_pad")
    await robot.release("RIGHT", "right_part", "right_pad")
    await robot.move("RIGHT", "right_clear")

    # Phase 2: publish own item-bound ready events.
    left_receipt = robot.signal("left_ready", "left_part")
    right_receipt = robot.signal("right_ready", "right_part")

    # Phase 3: each arm waits for peer ready, then consumes peer item.
    # LEFT consumes right_part from right_pad.
    left_peer = await robot.wait_event("right_ready", 5.0)
    await robot.move("LEFT", "left_pickup_wait")
    await robot.move("LEFT", "right_pad")
    await robot.grasp("LEFT", "right_part")
    await robot.move("LEFT", "left_target")
    robot.clear_event("right_ready", expected_version=left_peer.version)
    await robot.move("LEFT", "left_depart")

    # RIGHT consumes left_part from left_pad.
    right_peer = await robot.wait_event("left_ready", 5.0)
    await robot.move("RIGHT", "right_pickup_wait")
    await robot.move("RIGHT", "left_pad")
    await robot.grasp("RIGHT", "left_part")
    await robot.move("RIGHT", "right_target")
    robot.clear_event("left_ready", expected_version=right_peer.version)
    await robot.move("RIGHT", "right_depart")
