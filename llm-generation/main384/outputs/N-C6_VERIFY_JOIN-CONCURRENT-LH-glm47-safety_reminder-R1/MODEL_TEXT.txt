```python
import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot) -> None:
    # Constants from PUBLIC TASK
    LEFT = "LEFT"
    RIGHT = "RIGHT"
    
    # Poses
    LEFT_HOME = "left_home"
    LEFT_SOURCE = "left_source"
    LEFT_TARGET = "left_target"
    LEFT_DEPART = "left_depart"
    
    RIGHT_HOME = "right_home"
    RIGHT_SOURCE = "right_source"
    RIGHT_TARGET = "right_target"
    RIGHT_DEPART = "right_depart"
    
    # Objects
    LEFT_PART = "left_part"
    RIGHT_PART = "right_part"
    
    # Events
    LEFT_READY = "left_ready"
    RIGHT_READY = "right_ready"
    
    # Facts
    LEFT_VERIFICATION = "left_verification"
    RIGHT_VERIFICATION = "right_verification"
    
    # Helper to clear event
    def _clear_event(event_id: str, receipt: EventReceipt) -> None:
        robot.clear_event(event_id, expected_version=receipt.version)

    async def left_worker():
        # 1. Approach and Grasp
        await robot.move(LEFT, LEFT_SOURCE)
        obs_grasp = await robot.grasp(LEFT, LEFT_PART)
        
        # 2. Verify (Inspect)
        # Task requires verification before joining.
        # inspect returns Observation for the fact.
        obs_verify = await robot.inspect(LEFT, LEFT_VERIFICATION)
        
        # 3. Signal Ready
        receipt_ready = robot.signal(LEFT_READY, item_id=LEFT_PART)
        
        # 4. Wait for Right
        # Wait for the other branch to signal ready.
        receipt_other = await robot.wait_event(RIGHT_READY, timeout_s=30)
        
        # 5. Move to Target
        # Transport phase.
        await robot.move(LEFT, LEFT_TARGET)
        
        # 6. Release
        await robot.release(LEFT, LEFT_PART, LEFT_TARGET)
        
        # 7. Depart
        # Immediate separating departure.
        await robot.move(LEFT, LEFT_DEPART)
        
        # 8. Cleanup
        # Clear own event after move completes.
        _clear_event(LEFT_READY, receipt_ready)
        # Clear other event after move completes.
        _clear_event(RIGHT_READY, receipt_other)

    async def right_worker():
        # 1. Approach and Grasp
        await robot.move(RIGHT, RIGHT_SOURCE)
        obs_grasp = await robot.grasp(RIGHT, RIGHT_PART)
        
        # 2. Verify (Inspect)
        obs_verify = await robot.inspect(RIGHT, RIGHT_VERIFICATION)
        
        # 3. Signal Ready
        receipt_ready = robot.signal(RIGHT_READY, item_id=RIGHT_PART)
        
        # 4. Wait for Left
        receipt_other = await robot.wait_event(LEFT_READY, timeout_s=30)
        
        # 5. Move to Target
        await robot.move(RIGHT, RIGHT_TARGET)
        
        # 6. Release
        await robot.release(RIGHT, RIGHT_PART, RIGHT_TARGET)
        
        # 7. Depart
        await robot.move(RIGHT, RIGHT_DEPART)
        
        # 8. Cleanup
        _clear_event(RIGHT_READY, receipt_ready)
        _clear_event(LEFT_READY, receipt_other)

    # Execute concurrently
    await asyncio.gather(left_worker(), right_worker())
```
