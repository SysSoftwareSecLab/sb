import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault


async def run_task(robot: Robot) -> None:
    # SHORT variant: acquire tool only after any second-item wait and checks.
    # LEFT producer, RIGHT consumer, capacity-one buffer with item-specific contracts.
    # D4: join the two second-item checks with asyncio.gather inside i == 1 branch.

    async def producer() -> None:
        # Episode 0: part_0
        # LEFT picks part_0 from source_0 (approach from left_home)
        await robot.move("LEFT", "left_home")
        await robot.move("LEFT", "source_0")
        await robot.grasp("LEFT", "part_0")
        # Acquire buffer_lock before buffer entry
        await robot.acquire("LEFT", "buffer_lock", 5)
        # Move to buffer_0 (same physical location as buffer_1)
        await robot.move("LEFT", "buffer_0")
        await robot.release("LEFT", "part_0", "buffer_0")
        # Immediately depart before ready publication
        await robot.move("LEFT", "left_home")
        await robot.release_resource("LEFT", "buffer_lock")
        # Publish ready_0
        robot.signal("ready_0", "part_0")

        # Episode 1: part_1
        # Wait for empty_0 before entering buffer with second part
        await robot.wait_event("empty_0", 10)
        robot.clear_event("empty_0", expected_version=1)
        # SHORT: acquire tool only after second-item wait and checks
        await robot.acquire("LEFT", "tool", 5)
        # Second-item checks: inspect line_clear with LEFT, receiver_ready with RIGHT
        # D4: join them with asyncio.gather
        obs_line, obs_recv = await asyncio.gather(
            robot.inspect("LEFT", "line_clear"),
            robot.inspect("RIGHT", "receiver_ready"),
        )
        # Both public Boolean fields must permit transfer
        line_val = obs_line.value
        recv_val = obs_recv.value
        if not (isinstance(line_val, dict) and line_val.get("clear") is True):
            raise ContractError("line_clear does not permit transfer")
        if not (isinstance(recv_val, dict) and recv_val.get("ready") is True):
            raise ContractError("receiver_ready does not permit transfer")
        # Approach source_1 from left_wait
        await robot.move("LEFT", "left_wait")
        await robot.move("LEFT", "source_1")
        await robot.grasp("LEFT", "part_1")
        # Acquire buffer_lock before buffer entry
        await robot.acquire("LEFT", "buffer_lock", 5)
        await robot.move("LEFT", "buffer_1")
        await robot.release("LEFT", "part_1", "buffer_1")
        # Immediately depart before ready publication
        await robot.move("LEFT", "left_home")
        await robot.release_resource("LEFT", "buffer_lock")
        # Release tool on normal exit
        await robot.release_resource("LEFT", "tool")
        # Publish ready_1
        robot.signal("ready_1", "part_1")

    async def consumer() -> None:
        # Episode 0: part_0
        # Wait for ready_0 before pickup
        ready0 = await robot.wait_event("ready_0", 10)
        # Acquire buffer_lock before buffer entry
        await robot.acquire("RIGHT", "buffer_lock", 5)
        # Approach buffer_0 from right_home
        await robot.move("RIGHT", "right_home")
        await robot.move("RIGHT", "buffer_0")
        await robot.grasp("RIGHT", "part_0")
        # Depart buffer immediately
        await robot.move("RIGHT", "right_home")
        await robot.release_resource("RIGHT", "buffer_lock")
        # Carried move to target with exact active item receipt
        await robot.move("RIGHT", "target_0", receipt=ready0)
        await robot.release("RIGHT", "part_0", "target_0")
        # Depart target
        await robot.move("RIGHT", "right_home")
        # Clear ready_0 after carried move
        robot.clear_event("ready_0", expected_version=ready0.version)
        # Publish empty_0
        robot.signal("empty_0", "part_0")

        # Episode 1: part_1
        # Wait for ready_1 before pickup
        ready1 = await robot.wait_event("ready_1", 10)
        # Acquire buffer_lock before buffer entry
        await robot.acquire("RIGHT", "buffer_lock", 5)
        # Approach buffer_1 from right_wait
        await robot.move("RIGHT", "right_wait")
        await robot.move("RIGHT", "buffer_1")
        await robot.grasp("RIGHT", "part_1")
        # Depart buffer immediately
        await robot.move("RIGHT", "right_wait")
        await robot.release_resource("RIGHT", "buffer_lock")
        # Carried move to target with exact active item receipt
        await robot.move("RIGHT", "target_1", receipt=ready1)
        await robot.release("RIGHT", "part_1", "target_1")
        # Depart target
        await robot.move("RIGHT", "right_home")
        # Clear ready_1 after carried move
        robot.clear_event("ready_1", expected_version=ready1.version)
        # Publish empty_0 (second time)
        robot.signal("empty_0", "part_1")

    # B runs producer and consumer coroutines together
    await asyncio.gather(producer(), consumer())
