from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault
import asyncio


async def run_task(robot: Robot) -> None:
    # ------------------------------------------------------------------
    # Phase 1: each arm deposits its own part on its own pad, clears the
    # pad (moves away), then signals its own item-bound ready event.
    # ------------------------------------------------------------------

    async def deposit_own(arm: str, part: str, source: str, pad: str,
                          clear_pose: str, ready_event: str) -> None:
        # Approach own source from home, then grasp immediately.
        await robot.move(arm, source)
        await robot.grasp(arm, part)
        # Carry own part to own pad.
        await robot.move(arm, pad)
        # Release onto pad.
        await robot.release(arm, part, pad)
        # Immediately clear the pad (same coroutine, same virtual instant).
        await robot.move(arm, clear_pose)
        # Publish own item-bound ready event.
        robot.signal(ready_event, part)

    # ------------------------------------------------------------------
    # Phase 2: each arm waits for the peer's ready event, then picks up
    # the peer item from its pad, carries it to its own target, and
    # clears the peer event after the carried move completes.
    # ------------------------------------------------------------------

    async def consume_peer(arm: str, peer_part: str, peer_pad: str,
                           pickup_wait: str, target: str,
                           peer_ready_event: str) -> None:
        # Wait for peer ready event (must be active).
        receipt = await robot.wait_event(peer_ready_event, 10)
        # Move to the pickup-wait pose for the peer item's approach.
        await robot.move(arm, pickup_wait)
        # Approach the peer pad and grasp immediately.
        await robot.move(arm, peer_pad)
        await robot.grasp(arm, peer_part)
        # Carry the peer item to own target, consuming the receipt.
        await robot.move(arm, target, receipt=receipt)
        # Clear the peer event only after the carried move completed.
        robot.clear_event(peer_ready_event, expected_version=receipt.version)

    # ------------------------------------------------------------------
    # Structure: DEPENDENCY_DISTANCE = SHORT, layout L0.
    # Two workers run concurrently; each deposits then consumes.
    # ------------------------------------------------------------------

    left_worker = asyncio.create_task(
        _left_sequence(robot)
    )
    right_worker = asyncio.create_task(
        _right_sequence(robot)
    )

    await asyncio.gather(left_worker, right_worker)


async def _left_sequence(robot: Robot) -> None:
    # Deposit own part (left_part) on left_pad, clear, signal left_ready.
    await robot.move("LEFT", "left_source")
    await robot.grasp("LEFT", "left_part")
    await robot.move("LEFT", "left_pad")
    await robot.release("LEFT", "left_part", "left_pad")
    await robot.move("LEFT", "left_clear")
    robot.signal("left_ready", "left_part")

    # Consume peer part (right_part) from right_pad.
    receipt = await robot.wait_event("right_ready", 10)
    await robot.move("LEFT", "left_pickup_wait")
    await robot.move("LEFT", "right_pad")
    await robot.grasp("LEFT", "right_part")
    await robot.move("LEFT", "left_target", receipt=receipt)
    robot.clear_event("right_ready", expected_version=receipt.version)


async def _right_sequence(robot: Robot) -> None:
    # Deposit own part (right_part) on right_pad, clear, signal right_ready.
    await robot.move("RIGHT", "right_source")
    await robot.grasp("RIGHT", "right_part")
    await robot.move("RIGHT", "right_pad")
    await robot.release("RIGHT", "right_part", "right_pad")
    await robot.move("RIGHT", "right_clear")
    robot.signal("right_ready", "right_part")

    # Consume peer part (left_part) from left_pad.
    receipt = await robot.wait_event("left_ready", 10)
    await robot.move("RIGHT", "right_pickup_wait")
    await robot.move("RIGHT", "left_pad")
    await robot.grasp("RIGHT", "left_part")
    await robot.move("RIGHT", "right_target", receipt=receipt)
    robot.clear_event("left_ready", expected_version=receipt.version)
