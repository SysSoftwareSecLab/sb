from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt, ContractError, MotionFault
import asyncio


async def _worker(robot: Robot, arm: str, other_arm: str):
    # Each worker independently inspects allocation once before choosing branch.
    obs = await robot.inspect(arm, "allocation")
    value = obs.value
    selected = value["selected_arm"]
    if selected != arm:
        # Unselected arm makes no move/grasp/release call.
        return
    # Selected arm: approach from its home pose, then immediately grasp.
    home_pose = "left_home" if arm == "LEFT" else "right_home"
    await robot.move(arm, home_pose)
    await robot.grasp(arm, "shared_part", observation=obs)
    await robot.move(arm, "shared_target")
    await robot.release(arm, "shared_part", "shared_target")
    # Immediately depart.
    depart_pose = "left_depart" if arm == "LEFT" else "right_depart"
    await robot.move(arm, depart_pose)


async def run_task(robot: Robot):
    # Variant B: gather candidate workers concurrently.
    # A runs candidate LEFT then RIGHT serially; B gathers them concurrently.
    # Both perform the same inspections and one selected transport.
    left_task = asyncio.create_task(_worker(robot, "LEFT", "RIGHT"))
    right_task = asyncio.create_task(_worker(robot, "RIGHT", "LEFT"))
    await asyncio.gather(left_task, right_task)
