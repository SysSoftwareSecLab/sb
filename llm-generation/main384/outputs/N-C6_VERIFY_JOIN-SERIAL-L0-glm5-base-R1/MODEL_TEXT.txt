```python
import asyncio
from bridge_robot_api import Robot


async def _left_branch(robot: Robot) -> None:
    await robot.move("LEFT", "left_source")
    await robot.grasp("LEFT", "left_part")
    await robot.move("LEFT", "left_target")
    await robot.release("LEFT", "left_part", "left_target")
    await robot.move("LEFT", "left_depart")
    robot.signal("left_ready")


async def _right_branch(robot: Robot) -> None:
    await robot.move("RIGHT", "right_source")
    await robot.grasp("RIGHT", "right_part")
    await robot.move("RIGHT", "right_target")
    await robot.release("RIGHT", "right_part", "right_target")
    await robot.move("RIGHT", "right_depart")
    robot.signal("right_ready")


async def run_task(robot: Robot) -> None:
    await asyncio.gather(_left_branch(robot), _right_branch(robot))
    await robot.wait_event("left_ready", 5)
    await robot.wait_event("right_ready", 5)
    robot.clear_event("left_ready", expected_version=1)
    robot.clear_event("right_ready", expected_version=1)
```
