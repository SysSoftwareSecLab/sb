import asyncio

from bridge_robot_api import Robot


async def run_task(robot: Robot) -> None:
    # Phase 1: rq2_gate protocol (serial producer then consumer).
    gate_receipt = robot.signal("rq2_gate")
    await robot.acquire("LEFT", "rq2_gap_0", 5)
    await robot.release_resource("LEFT", "rq2_gap_0")
    await robot.acquire("LEFT", "rq2_gap_1", 5)
    await robot.release_resource("LEFT", "rq2_gap_1")
    await robot.acquire("LEFT", "rq2_gap_2", 5)
    await robot.release_resource("LEFT", "rq2_gap_2")
    gate_wait_receipt = await robot.wait_event("rq2_gate", 5)
    await robot.move("LEFT", "left_home", receipt=gate_wait_receipt)
    robot.clear_event("rq2_gate", expected_version=gate_receipt.version)

    # Phase 2: inherited dual-arm exchange, serial A order.
    # LEFT deposits left_part on left_pad, clears left_pad, signals left_ready.
    await robot.move("LEFT", "left_source")
    await robot.grasp("LEFT", "left_part")
    await robot.move("LEFT", "left_pad")
    await robot.release("LEFT", "left_part", "left_pad")
    await robot.move("LEFT", "left_pickup_wait")
    left_ready_receipt = robot.signal("left_ready")

    # RIGHT deposits right_part on right_pad, clears right_pad, signals right_ready.
    await robot.move("RIGHT", "right_source")
    await robot.grasp("RIGHT", "right_part")
    await robot.move("RIGHT", "right_pad")
    await robot.release("RIGHT", "right_part", "right_pad")
    await robot.move("RIGHT", "right_pickup_wait")
    right_ready_receipt = robot.signal("right_ready")

    # LEFT consumes right_part: wait for right_ready, carry its receipt to left_target, clear.
    left_consume_receipt = await robot.wait_event("right_ready", 5)
    await robot.move("LEFT", "right_pad", receipt=left_consume_receipt)
    await robot.grasp("LEFT", "right_part")
    await robot.move("LEFT", "left_target")
    await robot.release("LEFT", "right_part", "left_target")
    await robot.move("LEFT", "left_depart")
    robot.clear_event("right_ready", expected_version=right_ready_receipt.version)

    # RIGHT consumes left_part: wait for left_ready, carry its receipt to right_target, clear.
    right_consume_receipt = await robot.wait_event("left_ready", 5)
    await robot.move("RIGHT", "left_pad", receipt=right_consume_receipt)
    await robot.grasp("RIGHT", "left_part")
    await robot.move("RIGHT", "right_target")
    await robot.release("RIGHT", "left_part", "right_target")
    await robot.move("RIGHT", "right_depart")
    robot.clear_event("left_ready", expected_version=left_ready_receipt.version)
