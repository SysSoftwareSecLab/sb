import asyncio

from bridge_robot_api import MotionFault


async def run_task(robot):
    # ---- Phase 1: rq2_gate protocol (serial producer/consumer) ----
    for i in range(3):
        rid = f"rq2_gap_{i}"
        await robot.acquire("LEFT", rid, 5)
        await robot.release_resource("LEFT", rid)

    gate_receipt = robot.signal("rq2_gate")
    gate_wait = await robot.wait_event("rq2_gate", 5)
    clear_event_after_gate_wait(robot, gate_wait)
    clear_event_after_gate_signal(robot, gate_receipt)

    # ---- Phase 2: inherited dual-arm exchange (serial scheduling) ----
    # LEFT deposits left_part on left_pad, then clears left_pad.
    await robot.move("LEFT", "left_source")
    left_grasp_left = await robot.grasp("LEFT", "left_part")
    await robot.move("LEFT", "left_pad")
    await robot.release("LEFT", "left_part", "left_pad")
    await robot.move("LEFT", "left_clear")
    left_ready = robot.signal("left_ready", "left_part")

    # RIGHT deposits right_part on right_pad, then clears right_pad.
    await robot.move("RIGHT", "right_source")
    right_grasp_right = await robot.grasp("RIGHT", "right_part")
    await robot.move("RIGHT", "right_pad")
    await robot.release("RIGHT", "right_part", "right_pad")
    await robot.move("RIGHT", "right_clear")
    right_ready = robot.signal("right_ready", "right_part")

    # LEFT consumes right_part: wait for peer ready, approach, grasp, carry to target, clear.
    left_wait_right = await robot.wait_event("right_ready", 5)
    await robot.move("LEFT", "left_pickup_wait")
    await robot.grasp("LEFT", "right_part", observation=right_grasp_right)
    await robot.move("LEFT", "right_target", receipt=left_wait_right)
    clear_event_after_left_wait(robot, left_wait_right)
    await robot.release("LEFT", "right_part", "right_target")
    await robot.move("LEFT", "left_depart")
    clear_event_after_left_ready(robot, left_ready)

    # RIGHT consumes left_part: wait for peer ready, approach, grasp, carry to target, clear.
    right_wait_left = await robot.wait_event("left_ready", 5)
    await robot.move("RIGHT", "right_pickup_wait")
    await robot.grasp("RIGHT", "left_part", observation=left_grasp_left)
    await robot.move("RIGHT", "left_target", receipt=right_wait_left)
    clear_event_after_right_wait(robot, right_wait_left)
    await robot.release("RIGHT", "left_part", "left_target")
    await robot.move("RIGHT", "right_depart")
    clear_event_after_right_ready(robot, right_ready)


def clear_event_after_gate_wait(robot, receipt):
    robot.clear_event("rq2_gate", expected_version=receipt.version)


def clear_event_after_gate_signal(robot, receipt):
    robot.clear_event("rq2_gate", expected_version=receipt.version)


def clear_event_after_left_wait(robot, receipt):
    robot.clear_event("right_ready", expected_version=receipt.version)


def clear_event_after_left_ready(robot, receipt):
    robot.clear_event("left_ready", expected_version=receipt.version)


def clear_event_after_right_wait(robot, receipt):
    robot.clear_event("left_ready", expected_version=receipt.version)


def clear_event_after_right_ready(robot, receipt):
    robot.clear_event("right_ready", expected_version=receipt.version)
