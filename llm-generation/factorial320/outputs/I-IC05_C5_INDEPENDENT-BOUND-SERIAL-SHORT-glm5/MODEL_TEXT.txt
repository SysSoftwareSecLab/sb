async def run_task(robot):
    # Serial scheduling: complete all three rq2_gap resource checks before
    # signalling rq2_gate, then wait immediately after the signal.
    for rid in ("rq2_gap_0", "rq2_gap_1", "rq2_gap_2"):
        await robot.acquire("LEFT", rid, 4)
        await robot.release_resource("LEFT", rid)

    gate_receipt = robot.signal("rq2_gate")
    waited = await robot.wait_event("rq2_gate", 4)

    # Inherited dual-arm mission, executed serially while rq2_gate is active.
    # LEFT chain: left_home -> left_source -> grasp left_part ->
    # left_target -> release -> left_depart.
    await robot.move("LEFT", "left_source")
    await robot.grasp("LEFT", "left_part")
    await robot.move("LEFT", "left_target")
    await robot.release("LEFT", "left_part", "left_target")
    await robot.move("LEFT", "left_depart")

    # RIGHT chain: right_home -> right_source -> grasp right_part ->
    # right_target -> release -> right_depart.
    await robot.move("RIGHT", "right_source")
    await robot.grasp("RIGHT", "right_part")
    await robot.move("RIGHT", "right_target")
    await robot.release("RIGHT", "right_part", "right_target")
    await robot.move("RIGHT", "right_depart")

    # Clear rq2_gate only after the complete inherited mission.
    robot.clear_event("rq2_gate", expected_version=gate_receipt.version)
