```python
import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot) -> None:
    # Constants extracted from PUBLIC TASK
    LEFT = "LEFT"
    RIGHT = "RIGHT"
    
    # Poses
    LEFT_HOME = "left_home"
    LEFT_WAIT = "left_wait"
    RIGHT_HOME = "right_home"
    RIGHT_WAIT = "right_wait"
    
    SOURCE_0 = "source_0"
    SOURCE_1 = "source_1"
    BUFFER_0 = "buffer_0"
    BUFFER_1 = "buffer_1"
    TARGET_0 = "target_0"
    TARGET_1 = "target_1"
    
    # Items
    PART_0 = "part_0"
    PART_1 = "part_1"
    
    # Resources
    BUFFER_LOCK = "buffer_lock"
    
    # Events
    READY_0 = "ready_0"
    READY_1 = "ready_1"
    EMPTY_0 = "empty_0"
    RQ2_GATE = "rq2_gate"
    
    # Durations
    TIMEOUT_MOVE = 4.0
    TIMEOUT_EVENT = 50.0
    
    # Helper to run producer logic
    async def producer(item_id: str, source_pose: str, start_pose: str, ready_event: str, wait_event: str):
        # 1. Wait for empty_0 (except for first part where version is 0)
        if item_id == PART_1:
            empty_receipt = await robot.wait_event(EMPTY_0, TIMEOUT_EVENT)
            robot.clear_event(EMPTY_0, expected_version=empty_receipt.version)
        
        # 2. Acquire buffer_lock
        await robot.acquire(LEFT, BUFFER_LOCK, TIMEOUT_EVENT)
        
        # 3. Move to source (Approach)
        await robot.move(LEFT, start_pose, timeout_s=TIMEOUT_MOVE)
        # Immediate grasp
        await robot.grasp(LEFT, item_id)
        
        # 4. Move to buffer
        await robot.move(LEFT, source_pose, timeout_s=TIMEOUT_MOVE) # source_pose is the approach pose for the object
        await robot.move(LEFT, BUFFER_0 if item_id == PART_0 else BUFFER_1, timeout_s=TIMEOUT_MOVE)
        
        # 5. Release at buffer
        await robot.release(LEFT, item_id, BUFFER_0 if item_id == PART_0 else BUFFER_1)
        
        # 6. Depart buffer (Immediate separating departure)
        await robot.move(LEFT, start_pose, timeout_s=TIMEOUT_MOVE)
        
        # 7. Release buffer_lock
        await robot.release_resource(LEFT, BUFFER_LOCK)
        
        # 8. Signal ready
        robot.signal(ready_event, item_id=item_id)
        
        # 9. Wait for rq2_gate
        await robot.wait_event(RQ2_GATE, TIMEOUT_EVENT)

    # Helper to run consumer logic
    async def consumer(item_id: str, target_pose: str, start_pose: str, ready_event: str):
        # 1. Wait for ready event
        ready_receipt = await robot.wait_event(ready_event, TIMEOUT_EVENT)
        
        # 2. Acquire buffer_lock
        await robot.acquire(RIGHT, BUFFER_LOCK, TIMEOUT_EVENT)
        
        # 3. Move to buffer (Approach)
        await robot.move(RIGHT, start_pose, timeout_s=TIMEOUT_MOVE)
        # Immediate grasp
        await robot.grasp(RIGHT, item_id)
        
        # 4. Move to target using receipt
        await robot.move(RIGHT, target_pose, timeout_s=TIMEOUT_MOVE, receipt=ready_receipt)
        
        # 5. Clear ready event
        robot.clear_event(ready_event, expected_version=ready_receipt.version)
        
        # 6. Release at target
        await robot.release(RIGHT, item_id, target_pose)
        
        # 7. Depart target (Immediate separating departure)
        await robot.move(RIGHT, start_pose, timeout_s=TIMEOUT_MOVE)
        
        # 8. Release buffer_lock
        await robot.release_resource(RIGHT, BUFFER_LOCK)
        
        # 9. Signal empty_0
        robot.signal(EMPTY_0)

    # Main Loop: FOR -> PAR_JOIN -> consumer IF
    # Iteration 0: Part 0
    # Iteration 1: Part 1
    
    # --- Iteration 0 ---
    # PAR_JOIN: Producer and Consumer
    prod_0 = asyncio.create_task(producer(PART_0, SOURCE_0, LEFT_HOME, READY_0, EMPTY_0))
    cons_0 = asyncio.create_task(consumer(PART_0, TARGET_0, RIGHT_HOME, READY_0))
    
    # Wait for both to complete
    await prod_0
    await cons_0
    
    # --- Iteration 1 ---
    # PAR_JOIN: Producer and Consumer
    prod_1 = asyncio.create_task(producer(PART_1, SOURCE_1, LEFT_WAIT, READY_1, EMPTY_0))
    cons_1 = asyncio.create_task(consumer(PART_1, TARGET_1, RIGHT_WAIT, READY_1))
    
    # Wait for both to complete
    await prod_1
    await cons_1
    
    # Final state check: Ensure rq2_gate is inactive (cleared or not active)
    # The task requires "rq2_gate inactive at return".
    # Since we waited on it in producer, it might be active. 
    # However, the spec says "Inside one finite loop iteration... concurrently join an rq2_gate producer and consumer".
    # The producer waits for the gate. The consumer does not signal the gate.
    # The gate is signaled by the external system or implicitly?
    # The task says "rq2_gate inactive at return".
    # Given the producer waits for it, and we don't signal it, we assume it was signaled externally and we consumed it.
    # If it wasn't signaled, the producer would hang. Assuming valid execution path.
    # No explicit clear needed for rq2_gate in the goal string, just "inactive".
    # Since wait_event consumes the active state (returns receipt), it is no longer "active" in the sense of blocking?
    # Or does it need clearing? "program_clearable" is false for rq2_gate (not listed, default false).
    # So we just ensure we passed the wait.
    
    return
```
