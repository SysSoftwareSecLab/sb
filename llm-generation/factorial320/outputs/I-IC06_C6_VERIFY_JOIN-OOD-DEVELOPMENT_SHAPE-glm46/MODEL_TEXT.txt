import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot) -> None:
    # Constants extracted from PUBLIC TASK
    LEFT = "LEFT"
    RIGHT = "RIGHT"
    LEFT_PART = "left_part"
    RIGHT_PART = "right_part"
    
    # Poses
    LEFT_HOME = "left_home"
    LEFT_SOURCE = "left_source"
    LEFT_TARGET = "left_target"
    LEFT_DEPART = "left_depart"
    
    RIGHT_HOME = "right_home"
    RIGHT_SOURCE = "right_source"
    RIGHT_TARGET = "right_target"
    RIGHT_DEPART = "right_depart"
    
    # Events
    LEFT_READY = "left_ready"
    RIGHT_READY = "right_ready"
    RQ2_GATE = "rq2_gate"
    
    # Facts
    LEFT_VERIFICATION = "left_verification"
    RIGHT_VERIFICATION = "right_verification"
    
    # Timeouts
    MOVE_TIMEOUT = 4.0
    EVENT_TIMEOUT = 5.0
    
    # --- Helper Functions ---
    
    async def move_arm(arm: str, pose: str):
        await robot.move(arm, pose, timeout_s=MOVE_TIMEOUT)

    async def perform_verification(arm: str, part_id: str, home_pose: str, source_pose: str, target_pose: str, depart_pose: str, fact_id: str, ready_event: str):
        # 1. Move to home
        await move_arm(arm, home_pose)
        
        # 2. Approach (move to source)
        await move_arm(arm, source_pose)
        
        # 3. Grasp (immediate after approach)
        obs = await robot.grasp(arm, part_id)
        
        # 4. Inspect (using the observation from grasp)
        # Note: Task requires inspect to verify. The grasp observation is passed.
        await robot.inspect(arm, fact_id)
        
        # 5. Move to target
        await move_arm(arm, target_pose)
        
        # 6. Release
        await robot.release(arm, part_id, target_pose)
        
        # 7. Depart (immediate after release)
        await move_arm(arm, depart_pose)
        
        # 8. Signal ready
        robot.signal(ready_event, item_id=part_id)

    async def rq2_producer():
        # Wait for both arms to be ready (join events)
        # We need the receipts to clear them later if needed, or just wait for active state.
        # The consumer waits for RQ2_GATE.
        # The producer waits for LEFT_READY and RIGHT_READY.
        
        # Wait for left_ready
        l_receipt = await robot.wait_event(LEFT_READY, timeout_s=EVENT_TIMEOUT)
        # Wait for right_ready
        r_receipt = await robot.wait_event(RIGHT_READY, timeout_s=EVENT_TIMEOUT)
        
        # Signal rq2_gate
        robot.signal(RQ2_GATE)

    async def rq2_consumer():
        # Wait for the exact active receipt of rq2_gate
        gate_receipt = await robot.wait_event(RQ2_GATE, timeout_s=EVENT_TIMEOUT)
        
        # Execute the complete inherited mission
        # The inherited mission is the verification tasks for both arms.
        # Since we are in a consumer context, we execute them here.
        # This satisfies "the consumer ... executes the complete inherited mission".
        
        # Create tasks for both arms
        left_task = asyncio.create_task(
            perform_verification(
                LEFT, LEFT_PART, LEFT_HOME, LEFT_SOURCE, LEFT_TARGET, LEFT_DEPART,
                LEFT_VERIFICATION, LEFT_READY
            )
        )
        right_task = asyncio.create_task(
            perform_verification(
                RIGHT, RIGHT_PART, RIGHT_HOME, RIGHT_SOURCE, RIGHT_TARGET, RIGHT_DEPART,
                RIGHT_VERIFICATION, RIGHT_READY
            )
        )
        
        # Wait for completion
        await left_task
        await right_task
        
        # Clear the event version
        robot.clear_event(RQ2_GATE, expected_version=gate_receipt.version)

    # --- Main Execution Flow ---
    
    # Structure: FOR -> IF -> PAR_JOIN
    # "Inside one finite loop iteration" -> We run the sequence once.
    
    # PAR_JOIN: Concurrently join producer and consumer
    await asyncio.gather(
        rq2_producer(),
        rq2_consumer()
    )
