import asyncio
from bridge_robot_api import Robot, ActionReceipt, EventReceipt, MotionFault

async def run_task(robot: Robot):
    # Configuration constants derived from PUBLIC TASK
    ARMS = ["LEFT", "RIGHT"]
    RESOURCES = ["fixture", "tool"]
    GAP_RESOURCES = ["rq2_gap_0", "rq2_gap_1", "rq2_gap_2"]
    GATE_EVENT = "rq2_gate"
    
    TIMEOUT_S = 4.0
    ACQUIRE_TIMEOUT = 2.0
    
    # Poses
    HOME_POSES = {"LEFT": "left_home", "RIGHT": "right_home"}
    SOURCE_POSES = {"LEFT": "left_source", "RIGHT": "right_source"}
    TARGET_POSES = {"LEFT": "left_target", "RIGHT": "right_target"}
    DEPART_POSES = {"LEFT": "left_depart", "RIGHT": "right_depart"}
    
    OBJECTS = {"LEFT": "left_part", "RIGHT": "right_part"}

    async def acquire_resources(arm: str):
        """Acquire fixture then tool with safety cleanup."""
        acquired = []
        try:
            for res in RESOURCES:
                await robot.acquire(arm, res, ACQUIRE_TIMEOUT)
                acquired.append(res)
        except Exception:
            # Release any successfully acquired resources on failure
            for res in acquired:
                await robot.release_resource(arm, res)
            raise

    async def release_resources(arm: str):
        """Release tool then fixture."""
        # Release in reverse order of acquisition (tool then fixture)
        for res in reversed(RESOURCES):
            await robot.release_resource(arm, res)

    async def perform_dual_arm_mission():
        """Execute the inherited dual-arm mission concurrently."""
        
        async def worker(arm: str):
            # 1. Acquire resources
            await acquire_resources(arm)
            
            # 2. Move to source
            await robot.move(arm, SOURCE_POSES[arm], timeout_s=TIMEOUT_S)
            
            # 3. Grasp part
            await robot.grasp(arm, OBJECTS[arm])
            
            # 4. Move to target
            await robot.move(arm, TARGET_POSES[arm], timeout_s=TIMEOUT_S)
            
            # 5. Release part
            await robot.release(arm, OBJECTS[arm], TARGET_POSES[arm])
            
            # 6. Depart immediately
            await robot.move(arm, DEPART_POSES[arm], timeout_s=TIMEOUT_S)
            
            # 7. Release resources
            await release_resources(arm)

        # Run both workers concurrently
        await asyncio.gather(worker("LEFT"), worker("RIGHT"))

    async def gate_producer():
        """Handle rq2_gap checks and signal rq2_gate."""
        arm = "LEFT"
        
        # Acquire and release gaps in numeric order
        for res in GAP_RESOURCES:
            await robot.acquire(arm, res, ACQUIRE_TIMEOUT)
            await robot.release_resource(arm, res)
            
        # Signal the gate exactly once
        robot.signal(GATE_EVENT)

    async def gate_consumer():
        """Wait for rq2_gate and run the mission."""
        # Wait for the gate event
        await robot.wait_event(GATE_EVENT, TIMEOUT_S)
        
        # Run the complete inherited dual-arm mission
        await perform_dual_arm_mission()

    # --- Main Execution Flow ---
    
    # Launch producer and consumer concurrently
    producer_task = asyncio.create_task(gate_producer())
    consumer_task = asyncio.create_task(gate_consumer())
    
    # Wait for both to complete
    await asyncio.gather(producer_task, consumer_task)
    
    # Clear the event after the mission scope
    # We need the receipt from the wait_event to know the version.
    # Since wait_event returns the receipt, we must capture it in the consumer.
    # Refactoring to capture receipt:
    
    # Re-implementing flow to capture receipt for clearing
    async def controlled_flow():
        # 1. Launch Producer
        prod = asyncio.create_task(gate_producer())
        
        # 2. Launch Consumer (which waits and runs mission)
        # We need the receipt from wait_event inside the consumer logic
        receipt_holder = {"receipt": None}
        
        async def consumer_with_receipt():
            # Wait
            rcpt = await robot.wait_event(GATE_EVENT, TIMEOUT_S)
            receipt_holder["receipt"] = rcpt
            
            # Run Mission
            await perform_dual_arm_mission()
            
        cons = asyncio.create_task(consumer_with_receipt())
        
        # 3. Wait for both
        await asyncio.gather(prod, cons)
        
        # 4. Clear event using the captured receipt
        if receipt_holder["receipt"]:
            robot.clear_event(GATE_EVENT, expected_version=receipt_holder["receipt"].version)

    await controlled_flow()
