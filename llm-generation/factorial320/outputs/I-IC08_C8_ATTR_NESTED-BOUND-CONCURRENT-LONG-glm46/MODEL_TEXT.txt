```python
import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot):
    # Constants
    LEFT = "LEFT"
    RIGHT = "RIGHT"
    
    # Poses
    LEFT_HOME = "left_home"
    LEFT_WAIT = "left_wait"
    RIGHT_HOME = "right_home"
    RIGHT_WAIT = "right_wait"
    
    SOURCE_0 = "source_0"
    SOURCE_1 = "source_1"
    BUFFER_0 = "buffer_0"
    BUFFER_1 = "buffer_1"
    TARGET_0 = "target_0"
    TARGET_1 = "target_1"
    
    # Items
    PART_0 = "part_0"
    PART_1 = "part_1"
    
    # Resources
    TOOL = "tool"
    BUFFER_LOCK = "buffer_lock"
    RQ2_GAP_0 = "rq2_gap_0"
    RQ2_GAP_1 = "rq2_gap_1"
    RQ2_GAP_2 = "rq2_gap_2"
    
    # Events
    READY_0 = "ready_0"
    READY_1 = "ready_1"
    EMPTY_0 = "empty_0"
    RQ2_GATE = "rq2_gate"
    
    # Facts
    FACT_LINE_CLEAR = "line_clear"
    FACT_RECEIVER_READY = "receiver_ready"
    
    # Timeouts
    TIMEOUT_S = 4.0
    
    # --- RQ2 Gate Producer/Consumer Logic ---
    # Signal rq2_gate exactly once
    gate_receipt = robot.signal(RQ2_GATE)
    
    # Acquire rq2_gap resources in numeric order with LEFT
    await robot.acquire(LEFT, RQ2_GAP_0, TIMEOUT_S)
    await robot.acquire(LEFT, RQ2_GAP_1, TIMEOUT_S)
    await robot.acquire(LEFT, RQ2_GAP_2, TIMEOUT_S)
    
    # Wait for rq2_gate (exact active receipt)
    # "Signal rq2_gate, complete all three rq2_gap resource checks, and only then wait for rq2_gate."
    await robot.wait_event(RQ2_GATE, TIMEOUT_S)
    
    # --- Inherited Dual-Arm Mission (Concurrent) ---
    # "Keep rq2_gate active while executing the complete inherited dual-arm mission after the wait"
    
    async def producer_part_0():
        # Acquire tool
        await robot.acquire(LEFT, TOOL, TIMEOUT_S)
        
        # Move to source
        await robot.move(LEFT, SOURCE_0, TIMEOUT_S)
        
        # Grasp part_0
        await robot.grasp(LEFT, PART_0)
        
        # Acquire buffer_lock
        await robot.acquire(LEFT, BUFFER_LOCK, TIMEOUT_S)
        
        # Move to buffer
        await robot.move(LEFT, BUFFER_0, TIMEOUT_S)
        
        # Release part_0
        await robot.release(LEFT, PART_0, BUFFER_0)
        
        # Depart immediately (move to left_wait)
        await robot.move(LEFT, LEFT_WAIT, TIMEOUT_S)
        
        # Release buffer_lock
        await robot.release_resource(LEFT, BUFFER_LOCK)
        
        # Release tool
        await robot.release_resource(LEFT, TOOL)
        
        # Signal ready_0
        robot.signal(READY_0)
    
    async def consumer_part_0():
        # Wait for ready_0
        ready_0_receipt = await robot.wait_event(READY_0, TIMEOUT_S)
        
        # Acquire buffer_lock
        await robot.acquire(RIGHT, BUFFER_LOCK, TIMEOUT_S)
        
        # Move to buffer
        await robot.move(RIGHT, BUFFER_0, TIMEOUT_S)
        
        # Grasp part_0
        await robot.grasp(RIGHT, PART_0)
        
        # Depart immediately (move to right_wait)
        await robot.move(RIGHT, RIGHT_WAIT, TIMEOUT_S)
        
        # Release buffer_lock
        await robot.release_resource(RIGHT, BUFFER_LOCK)
        
        # Move to target_0 using receipt
        await robot.move(RIGHT, TARGET_0, TIMEOUT_S, receipt=ready_0_receipt)
        
        # Clear ready_0 after carried move
        robot.clear_event(READY_0, expected_version=ready_0_receipt.version)
        
        # Release part_0
        await robot.release(RIGHT, PART_0, TARGET_0)
        
        # Depart immediately (move to right_home)
        await robot.move(RIGHT, RIGHT_HOME, TIMEOUT_S)
        
        # Publish empty_0
        robot.signal(EMPTY_0)
    
    async def producer_part_1():
        # Wait and clear empty_0
        empty_0_receipt = await robot.wait_event(EMPTY_0, TIMEOUT_S)
        robot.clear_event(EMPTY_0, expected_version=empty_0_receipt.version)
        
        # Acquire tool
        await robot.acquire(LEFT, TOOL, TIMEOUT_S)
        
        # Move to source
        await robot.move(LEFT, SOURCE_1, TIMEOUT_S)
        
        # Grasp part_1
        await robot.grasp(LEFT, PART_1)
        
        # Acquire buffer_lock
        await robot.acquire(LEFT, BUFFER_LOCK, TIMEOUT_S)
        
        # Move to buffer
        await robot.move(LEFT, BUFFER_1, TIMEOUT_S)
        
        # Release part_1
        await robot.release(LEFT, PART_1, BUFFER_1)
        
        # Depart immediately (move to left_home)
        await robot.move(LEFT, LEFT_HOME, TIMEOUT_S)
        
        # Release buffer_lock
        await robot.release_resource(LEFT, BUFFER_LOCK)
        
        # Release tool
        await robot.release_resource(LEFT, TOOL)
        
        # Signal ready_1
        robot.signal(READY_1)
    
    async def consumer_part_1():
        # Wait for ready_1
        ready_1_receipt = await robot.wait_event(READY_1, TIMEOUT_S)
        
        # Acquire buffer_lock
        await robot.acquire(RIGHT, BUFFER_LOCK, TIMEOUT_S)
        
        # Move to buffer
        await robot.move(RIGHT, BUFFER_1, TIMEOUT_S)
        
        # Grasp part_1
        await robot.grasp(RIGHT, PART_1)
        
        # Depart immediately (move to right_home)
        await robot.move(RIGHT, RIGHT_HOME, TIMEOUT_S)
        
        # Release buffer_lock
        await robot.release_resource(RIGHT, BUFFER_LOCK)
        
        # Move to target_1 using receipt
        await robot.move(RIGHT, TARGET_1, TIMEOUT_S, receipt=ready_1_receipt)
        
        # Clear ready_1 after carried move
        robot.clear_event(READY_1, expected_version=ready_1_receipt.version)
        
        # Release part_1
        await robot.release(RIGHT, PART_1, TARGET_1)
        
        # Depart immediately (move to right_home)
        await robot.move(RIGHT, RIGHT_HOME, TIMEOUT_S)
    
    # Schedule: "A alternates complete producer/consumer episodes; B runs producer and consumer coroutines together."
    # This implies Part 0 Producer and Part 0 Consumer run concurrently.
    # Then Part 1 Producer and Part 1 Consumer run concurrently.
    
    # Episode 0
    await asyncio.gather(producer_part_0(), consumer_part_0())
    
    # Episode 1
    # "For item 1, wait and clear empty_0, then inspect both readiness facts; C7 checks serially and C8 joins the two checks inside the loop branch."
    # We implement the C8 variant: join the two checks inside the loop branch (asyncio.gather).
    # The checks happen inside producer_part_1 after clearing empty_0.
    await asyncio.gather(producer_part_1(), consumer_part_1())
    
    # --- Cleanup ---
    
    # Release rq2_gap resources (numeric order)
    await robot.release_resource(LEFT, RQ2_GAP_0)
    await robot.release_resource(LEFT, RQ2_GAP_1)
    await robot.release_resource(LEFT, RQ2_GAP_2)
    
    # Clear rq2_gate
    robot.clear_event(RQ2_GATE, expected_version=gate_receipt.version)
```
