import asyncio

async def run_task(robot):
    # Constants
    LEFT = "LEFT"
    RIGHT = "RIGHT"
    
    # Poses
    LEFT_HOME = "left_home"
    RIGHT_HOME = "right_home"
    LEFT_SOURCE = "left_source"
    RIGHT_SOURCE = "right_source"
    LEFT_PAD = "left_pad"
    RIGHT_PAD = "right_pad"
    LEFT_TARGET = "left_target"
    RIGHT_TARGET = "right_target"
    LEFT_DEPART = "left_depart"
    RIGHT_DEPART = "right_depart"
    LEFT_PICKUP_WAIT = "left_pickup_wait"
    RIGHT_PICKUP_WAIT = "right_pickup_wait"
    
    # Objects
    LEFT_PART = "left_part"
    RIGHT_PART = "right_part"
    
    # Events
    LEFT_READY = "left_ready"
    RIGHT_READY = "right_ready"
    RQ2_GATE = "rq2_gate"
    
    # Resources
    GAP_0 = "rq2_gap_0"
    GAP_1 = "rq2_gap_1"
    GAP_2 = "rq2_gap_2"
    
    # Timeouts
    TIMEOUT_S = 4.0
    
    # --- Phase 1: Resource Acquisition (Serial) ---
    # Acquire rq2_gap_0, rq2_gap_1, rq2_gap_2 with LEFT in numeric order
    await robot.acquire(LEFT, GAP_0, TIMEOUT_S)
    await robot.acquire(LEFT, GAP_1, TIMEOUT_S)
    await robot.acquire(LEFT, GAP_2, TIMEOUT_S)
    
    # --- Phase 2: Signal RQ2_GATE ---
    # Signal rq2_gate exactly once
    gate_receipt = robot.signal(RQ2_GATE)
    
    # --- Phase 3: Wait for RQ2_GATE ---
    # Wait for its exact active receipt exactly once
    # "Signal rq2_gate, complete all three rq2_gap resource checks, and only then wait for rq2_gate."
    # The checks (acquire) are done. Now wait.
    active_gate_receipt = await robot.wait_event(RQ2_GATE, TIMEOUT_S)
    
    # --- Phase 4: Inherited Dual-Arm Mission (Serial) ---
    # "Use serial scheduling for both the inherited dual-arm mission"
    # Variant A: deposit LEFT then RIGHT, then consume LEFT then RIGHT.
    
    # 1. LEFT deposits its own part (left_part) on left_pad
    # Approach: left_home -> left_source
    await robot.move(LEFT, LEFT_SOURCE, timeout_s=TIMEOUT_S)
    # Grasp left_part
    await robot.grasp(LEFT, LEFT_PART)
    # Move to left_pad
    await robot.move(LEFT, LEFT_PAD, timeout_s=TIMEOUT_S)
    # Release left_part on left_pad
    await robot.release(LEFT, LEFT_PART, LEFT_PAD)
    # Immediately clear the pad (depart)
    await robot.move(LEFT, LEFT_DEPART, timeout_s=TIMEOUT_S)
    
    # Publish LEFT ready event
    left_ready_receipt = robot.signal(LEFT_READY)
    
    # 2. RIGHT deposits its own part (right_part) on right_pad
    # Approach: right_home -> right_source
    await robot.move(RIGHT, RIGHT_SOURCE, timeout_s=TIMEOUT_S)
    # Grasp right_part
    await robot.grasp(RIGHT, RIGHT_PART)
    # Move to right_pad
    await robot.move(RIGHT, RIGHT_PAD, timeout_s=TIMEOUT_S)
    # Release right_part on right_pad
    await robot.release(RIGHT, RIGHT_PART, RIGHT_PAD)
    # Immediately clear the pad (depart)
    await robot.move(RIGHT, RIGHT_DEPART, timeout_s=TIMEOUT_S)
    
    # Publish RIGHT ready event
    right_ready_receipt = robot.signal(RIGHT_READY)
    
    # 3. LEFT consumes peer item (right_part)
    # "Before an arm picks up the peer item, both own deposit/clear and a wait for peer ready must have completed."
    # Wait for RIGHT_READY
    peer_right_ready = await robot.wait_event(RIGHT_READY, TIMEOUT_S)
    
    # Move to pickup wait pose
    await robot.move(LEFT, LEFT_PICKUP_WAIT, timeout_s=TIMEOUT_S)
    # Approach: left_pickup_wait -> right_pad
    await robot.move(LEFT, RIGHT_PAD, timeout_s=TIMEOUT_S)
    # Grasp right_part
    await robot.grasp(LEFT, RIGHT_PART)
    
    # "The carried move to its own target must carry the exact active peer-item receipt returned by wait"
    # Move to left_target carrying the receipt
    await robot.move(LEFT, LEFT_TARGET, timeout_s=TIMEOUT_S, receipt=peer_right_ready)
    
    # Release right_part on left_target
    await robot.release(LEFT, RIGHT_PART, LEFT_TARGET)
    # Depart
    await robot.move(LEFT, LEFT_DEPART, timeout_s=TIMEOUT_S)
    
    # Clear peer event after the move
    robot.clear_event(RIGHT_READY, expected_version=peer_right_ready.version)
    
    # 4. RIGHT consumes peer item (left_part)
    # Wait for LEFT_READY
    peer_left_ready = await robot.wait_event(LEFT_READY, TIMEOUT_S)
    
    # Move to pickup wait pose
    await robot.move(RIGHT, RIGHT_PICKUP_WAIT, timeout_s=TIMEOUT_S)
    # Approach: right_pickup_wait -> left_pad
    await robot.move(RIGHT, LEFT_PAD, timeout_s=TIMEOUT_S)
    # Grasp left_part
    await robot.grasp(RIGHT, LEFT_PART)
    
    # Move to right_target carrying the receipt
    await robot.move(RIGHT, RIGHT_TARGET, timeout_s=TIMEOUT_S, receipt=peer_left_ready)
    
    # Release left_part on right_target
    await robot.release(RIGHT, LEFT_PART, RIGHT_TARGET)
    # Depart
    await robot.move(RIGHT, RIGHT_DEPART, timeout_s=TIMEOUT_S)
    
    # Clear peer event after the move
    robot.clear_event(LEFT_READY, expected_version=peer_left_ready.version)
    
    # --- Phase 5: Cleanup ---
    # "Keep rq2_gate active while executing the complete inherited dual-arm mission after the wait; clear only after the mission."
    # Mission is done. Clear RQ2_GATE.
    robot.clear_event(RQ2_GATE, expected_version=active_gate_receipt.version)
    
    # Release resources (LEFT arm holds them)
    # "never retain them at return"
    await robot.release_resource(LEFT, GAP_0)
    await robot.release_resource(LEFT, GAP_1)
    await robot.release_resource(LEFT, GAP_2)
