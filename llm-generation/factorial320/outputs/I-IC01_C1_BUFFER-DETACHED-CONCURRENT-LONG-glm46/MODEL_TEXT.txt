import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot):
    # RQ2_GATE PROTOCOL
    # 1. Signal rq2_gate
    gate_receipt = robot.signal("rq2_gate")
    
    # 2. Acquire and release rq2_gap_0, rq2_gap_1, rq2_gap_2 with LEFT in numeric order
    #    Must be OFF at return, so we release them immediately.
    #    Timeout 4s is safe within 50s deadline.
    gap_timeout = 4.0
    
    await robot.acquire("LEFT", "rq2_gap_0", gap_timeout)
    await robot.release_resource("LEFT", "rq2_gap_0")
    
    await robot.acquire("LEFT", "rq2_gap_1", gap_timeout)
    await robot.release_resource("LEFT", "rq2_gap_1")
    
    await robot.acquire("LEFT", "rq2_gap_2", gap_timeout)
    await robot.release_resource("LEFT", "rq2_gap_2")
    
    # 3. Wait for rq2_gate (exact active receipt)
    await robot.wait_event("rq2_gate", gap_timeout)
    
    # 4. Clear rq2_gate (exact version)
    robot.clear_event("rq2_gate", expected_version=gate_receipt.version)

    # INHERITED DUAL-ARM MISSION (Variant B: Concurrent Producer/Consumer)
    # LEFT: Producer (source -> buffer)
    # RIGHT: Consumer (buffer -> target)
    
    # Episode 1: part_0
    async def producer_part_0():
        # Move to source_0 (approach start)
        await robot.move("LEFT", "source_0")
        # Grasp part_0
        await robot.grasp("LEFT", "part_0")
        
        # Acquire buffer_lock
        await robot.acquire("LEFT", "buffer_lock", 4.0)
        
        # Move to buffer_0
        await robot.move("LEFT", "buffer_0")
        # Release part_0 at buffer_0
        await robot.release("LEFT", "part_0", "buffer_0")
        
        # Depart buffer immediately (move to left_wait)
        await robot.move("LEFT", "left_wait")
        
        # Release buffer_lock
        await robot.release_resource("LEFT", "buffer_lock")
        
        # Signal ready_0
        robot.signal("ready_0")

    async def consumer_part_0():
        # Wait for ready_0
        ready_receipt = await robot.wait_event("ready_0", 4.0)
        
        # Acquire buffer_lock
        await robot.acquire("RIGHT", "buffer_lock", 4.0)
        
        # Move to buffer_0 (approach start)
        await robot.move("RIGHT", "buffer_0")
        # Grasp part_0
        await robot.grasp("RIGHT", "part_0")
        
        # Depart buffer immediately (move to right_wait)
        await robot.move("RIGHT", "right_wait")
        
        # Release buffer_lock
        await robot.release_resource("RIGHT", "buffer_lock")
        
        # Move to target_0 using the ready receipt
        await robot.move("RIGHT", "target_0", receipt=ready_receipt)
        
        # Clear ready_0 after carried move
        robot.clear_event("ready_0", expected_version=ready_receipt.version)
        
        # Release part_0 at target_0
        await robot.release("RIGHT", "part_0", "target_0")
        
        # Depart target immediately (move to right_home)
        await robot.move("RIGHT", "right_home")
        
        # Signal empty_0
        robot.signal("empty_0")

    # Run Episode 1 concurrently
    await asyncio.gather(producer_part_0(), consumer_part_0())

    # Episode 2: part_1
    async def producer_part_1():
        # Wait for empty_0 before entering buffer with second part
        empty_receipt = await robot.wait_event("empty_0", 4.0)
        # Clear empty_0
        robot.clear_event("empty_0", expected_version=empty_receipt.version)
        
        # Move to source_1 (approach start)
        await robot.move("LEFT", "source_1")
        # Grasp part_1
        await robot.grasp("LEFT", "part_1")
        
        # Acquire buffer_lock
        await robot.acquire("LEFT", "buffer_lock", 4.0)
        
        # Move to buffer_1
        await robot.move("LEFT", "buffer_1")
        # Release part_1 at buffer_1
        await robot.release("LEFT", "part_1", "buffer_1")
        
        # Depart buffer immediately (move to left_home)
        await robot.move("LEFT", "left_home")
        
        # Release buffer_lock
        await robot.release_resource("LEFT", "buffer_lock")
        
        # Signal ready_1
        robot.signal("ready_1")

    async def consumer_part_1():
        # Wait for ready_1
        ready_receipt = await robot.wait_event("ready_1", 4.0)
        
        # Acquire buffer_lock
        await robot.acquire("RIGHT", "buffer_lock", 4.0)
        
        # Move to buffer_1 (approach start)
        await robot.move("RIGHT", "buffer_1")
        # Grasp part_1
        await robot.grasp("RIGHT", "part_1")
        
        # Depart buffer immediately (move to right_home)
        await robot.move("RIGHT", "right_home")
        
        # Release buffer_lock
        await robot.release_resource("RIGHT", "buffer_lock")
        
        # Move to target_1 using the ready receipt
        await robot.move("RIGHT", "target_1", receipt=ready_receipt)
        
        # Clear ready_1 after carried move
        robot.clear_event("ready_1", expected_version=ready_receipt.version)
        
        # Release part_1 at target_1
        await robot.release("RIGHT", "part_1", "target_1")
        
        # Depart target immediately (already at right_home, but ensure move if needed or just finish)
        # Goal says "empty arms at homes". Right is at home. Left is at home.
        # No further action needed for RIGHT.

    # Run Episode 2 concurrently
    await asyncio.gather(producer_part_1(), consumer_part_1())
