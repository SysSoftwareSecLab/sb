import asyncio
from bridge_robot_api import Robot


async def run_task(robot: Robot) -> None:
    # ---- RQ2 gate protocol (serial, before inherited dual-arm mission) ----
    gate_receipt = robot.signal("rq2_gate")
    await robot.wait_event("rq2_gate", 5)

    for gap in ("rq2_gap_0", "rq2_gap_1", "rq2_gap_2"):
        await robot.acquire("LEFT", gap, 5)
        await robot.release_resource("LEFT", gap)

    robot.clear_event("rq2_gate", expected_version=gate_receipt.version)

    # ---- Inherited dual-arm buffer mission, serial scheduling ----
    await _episode_part_0(robot)
    await _episode_part_1(robot)


async def _episode_part_0(robot: Robot) -> None:
    # Producer (LEFT): source_0 -> buffer_0, depart, publish ready_0.
    await robot.acquire("LEFT", "buffer_lock", 5)
    await robot.move("LEFT", "source_0")
    await robot.grasp("LEFT", "part_0")
    await robot.move("LEFT", "buffer_0", receipt=None)
    await robot.release("LEFT", "part_0", "buffer_0")
    await robot.move("LEFT", "left_home")
    await robot.release_resource("LEFT", "buffer_lock")
    ready_receipt = robot.signal("ready_0", "part_0")

    # Consumer (RIGHT): wait ready_0, carry to target_0 using that receipt,
    # clear ready_0, release on target_0, depart, publish empty_0.
    await robot.wait_event("ready_0", 5)
    await robot.acquire("RIGHT", "buffer_lock", 5)
    await robot.move("RIGHT", "buffer_0")
    await robot.grasp("RIGHT", "part_0")
    await robot.move("RIGHT", "target_0", receipt=ready_receipt)
    robot.clear_event("ready_0", expected_version=ready_receipt.version)
    await robot.release("RIGHT", "part_0", "target_0")
    await robot.move("RIGHT", "right_home")
    await robot.release_resource("RIGHT", "buffer_lock")
    empty_receipt = robot.signal("empty_0", "part_0")

    # Producer waits and clears empty_0 before entering buffer with second part.
    await robot.wait_event("empty_0", 5)
    robot.clear_event("empty_0", expected_version=empty_receipt.version)


async def _episode_part_1(robot: Robot) -> None:
    # Producer (LEFT): owns tool through source pickup and ready publication,
    # releases tool on every exit. left_home -> left_wait -> source_1.
    await robot.acquire("LEFT", "tool", 5)
    await robot.move("LEFT", "left_wait")
    await robot.move("LEFT", "source_1")
    await robot.grasp("LEFT", "part_1")
    await robot.move("LEFT", "left_wait")
    await robot.release_resource("LEFT", "tool")

    await robot.acquire("LEFT", "buffer_lock", 5)
    await robot.acquire("LEFT", "tool", 5)
    await robot.move("LEFT", "buffer_1")
    await robot.release("LEFT", "part_1", "buffer_1")
    await robot.move("LEFT", "left_wait")
    await robot.release_resource("LEFT", "tool")
    await robot.release_resource("LEFT", "buffer_lock")
    ready_receipt = robot.signal("ready_1", "part_1")

    # Consumer (RIGHT): wait ready_1, carry to target_1 using that receipt,
    # clear ready_1, release on target_1, depart, publish empty_0.
    await robot.wait_event("ready_1", 5)
    await robot.acquire("RIGHT", "buffer_lock", 5)
    await robot.move("RIGHT", "right_wait")
    await robot.move("RIGHT", "buffer_1")
    await robot.grasp("RIGHT", "part_1")
    await robot.move("RIGHT", "target_1", receipt=ready_receipt)
    robot.clear_event("ready_1", expected_version=ready_receipt.version)
    await robot.release("RIGHT", "part_1", "target_1")
    await robot.move("RIGHT", "right_home")
    await robot.release_resource("RIGHT", "buffer_lock")
    empty_receipt = robot.signal("empty_0", "part_1")

    # Producer waits and clears empty_0.
    await robot.wait_event("empty_0", 5)
    robot.clear_event("empty_0", expected_version=empty_receipt.version)
