import asyncio
from bridge_robot_api import MotionFault


async def run_task(robot):
    """IC03_C3_EXCHANGE-BOUND-CONCURRENT-LONG, variant B.

    Two concurrent workers each deposit their own part, clear their pad, and
    publish their own item-bound ready event.  After waiting for the peer ready,
    each worker picks up the peer item, carries it to its own target using the
    exact active peer-item receipt returned by wait, then clears that event.
    The rq2_gate producer/consumer is joined to the same concurrent schedule:
    signal rq2_gate, acquire/release rq2_gap_0..2 in numeric order with LEFT,
    then wait for rq2_gate, keep it active through the inherited dual-arm
    mission, and clear it only after the mission.  rq2_gate is cleared exactly
    once with the exact version returned by wait.
    """

    async def left_worker(gate_receipt_holder):
        # --- Deposit LEFT's own part on left_pad ---
        await robot.move("LEFT", "left_home")
        await robot.move("LEFT", "left_source")
        await robot.grasp("LEFT", "left_part")
        await robot.move("LEFT", "left_pad")
        await robot.release("LEFT", "left_part", "left_pad")
        await robot.move("LEFT", "left_clear")
        left_ready_receipt = robot.signal("left_ready", "left_part")

        # --- Wait for peer ready before consuming peer item ---
        right_ready_receipt = await robot.wait_event("right_ready", 30)

        # --- Pick up peer (right) part ---
        await robot.move("LEFT", "left_pickup_wait")
        await robot.move("LEFT", "right_pad")
        await robot.grasp("LEFT", "right_part")
        await robot.move("LEFT", "left_pickup_wait")

        # --- Carry peer item to own target using exact active peer receipt ---
        await robot.move("LEFT", "left_target", receipt=right_ready_receipt)
        await robot.release("LEFT", "right_part", "left_target")
        await robot.move("LEFT", "left_depart")

        # --- Clear the peer-item event after the carrying move completed ---
        robot.clear_event("right_ready", expected_version=right_ready_receipt.version)

        # --- Clear own ready event (now inactive at return) ---
        robot.clear_event("left_ready", expected_version=left_ready_receipt.version)

        # --- rq2_gate producer/consumer joined to this worker ---
        gate_receipt_holder["receipt"] = await robot.wait_event("rq2_gate", 30)

        await robot.acquire("LEFT", "rq2_gap_0", 30)
        await robot.release_resource("LEFT", "rq2_gap_0")
        await robot.acquire("LEFT", "rq2_gap_1", 30)
        await robot.release_resource("LEFT", "rq2_gap_1")
        await robot.acquire("LEFT", "rq2_gap_2", 30)
        await robot.release_resource("LEFT", "rq2_gap_2")

        # Clear rq2_gate exactly once with the exact version returned by wait.
        robot.clear_event("rq2_gate", expected_version=gate_receipt_holder["receipt"].version)

    async def right_worker():
        # --- Deposit RIGHT's own part on right_pad ---
        await robot.move("RIGHT", "right_home")
        await robot.move("RIGHT", "right_source")
        await robot.grasp("RIGHT", "right_part")
        await robot.move("RIGHT", "right_pad")
        await robot.release("RIGHT", "right_part", "right_pad")
        await robot.move("RIGHT", "right_clear")
        right_ready_receipt = robot.signal("right_ready", "right_part")

        # --- Wait for peer ready before consuming peer item ---
        left_ready_receipt = await robot.wait_event("left_ready", 30)

        # --- Pick up peer (left) part ---
        await robot.move("RIGHT", "right_pickup_wait")
        await robot.move("RIGHT", "left_pad")
        await robot.grasp("RIGHT", "left_part")
        await robot.move("RIGHT", "right_pickup_wait")

        # --- Carry peer item to own target using exact active peer receipt ---
        await robot.move("RIGHT", "right_target", receipt=left_ready_receipt)
        await robot.release("RIGHT", "left_part", "right_target")
        await robot.move("RIGHT", "right_depart")

        # --- Clear the peer-item event after the carrying move completed ---
        robot.clear_event("left_ready", expected_version=left_ready_receipt.version)

        # --- Clear own ready event (now inactive at return) ---
        robot.clear_event("right_ready", expected_version=right_ready_receipt.version)

    gate_receipt_holder = {"receipt": None}

    # Signal rq2_gate exactly once before the inherited dual-arm mission.
    robot.signal("rq2_gate")

    # Joined concurrent scheduling for both the inherited dual-arm mission and
    # the rq2_gate producer/consumer.
    await asyncio.gather(
        left_worker(gate_receipt_holder),
        right_worker(),
    )
