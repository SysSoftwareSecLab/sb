import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot):
    # --- RQ2 Gate Protocol ---
    # 1. Signal rq2_gate
    gate_receipt = robot.signal("rq2_gate")
    
    # 2. Acquire and release rq2_gap_0, rq2_gap_1, rq2_gap_2 with LEFT in numeric order
    for res_id in ["rq2_gap_0", "rq2_gap_1", "rq2_gap_2"]:
        await robot.acquire("LEFT", res_id, 10.0)
        await robot.release_resource("LEFT", res_id)
    
    # 3. Wait for rq2_gate (exact active receipt)
    await robot.wait_event("rq2_gate", 10.0)
    
    # 4. Clear rq2_gate (exact version)
    robot.clear_event("rq2_gate", expected_version=gate_receipt.version)

    # --- Inherited Dual-Arm Mission (Concurrent) ---
    
    # Helper for Producer (LEFT)
    async def producer(item_id: str, source_pose: str, ready_event_id: str, tool_res: str):
        # 1. Acquire tool
        await robot.acquire("LEFT", tool_res, 10.0)
        
        # 2. Move to source and grasp
        await robot.move("LEFT", source_pose)
        obs = await robot.grasp("LEFT", item_id)
        
        # 3. Move to buffer (acquire lock)
        await robot.acquire("LEFT", "buffer_lock", 10.0)
        await robot.move("LEFT", "buffer_0")
        
        # 4. Release at buffer
        await robot.release("LEFT", item_id, "buffer_0")
        
        # 5. Depart buffer (immediate departure)
        await robot.move("LEFT", "left_home")
        await robot.release_resource("LEFT", "buffer_lock")
        
        # 6. Release tool
        await robot.release_resource("LEFT", tool_res)
        
        # 7. Signal ready
        robot.signal(ready_event_id, item_id=item_id)

    # Helper for Consumer (RIGHT)
    async def consumer(item_id: str, target_pose: str, ready_event_id: str):
        # 1. Wait for ready
        ready_receipt = await robot.wait_event(ready_event_id, 10.0)
        
        # 2. Move to buffer (acquire lock)
        await robot.acquire("RIGHT", "buffer_lock", 10.0)
        await robot.move("RIGHT", "buffer_0")
        
        # 3. Grasp (using observation from ready receipt if needed, but spec says wait receipt)
        # Note: grasp requires observation if specified by task. 
        # Task says "Consumer waits the corresponding ready receipt before pickup, and supplies that exact active item receipt on carried move".
        # It does not explicitly say grasp needs the receipt, but grasp needs observation if task says so.
        # Given "wait and clear empty_0, then inspect both readiness facts" for item 1, 
        # and general grasp rules, we pass the receipt as observation if valid, or None.
        # Assuming the receipt acts as the observation credential here or is not strictly required for grasp itself but for move.
        # However, standard grasp often needs observation. Let's assume the receipt is the observation credential.
        # Wait, Observation is returned by grasp/inspect. EventReceipt is returned by wait_event.
        # The spec says "supplies that exact active item receipt on carried move". It doesn't say grasp.
        # But grasp requires observation if task says. "Consumer waits... before pickup".
        # Let's try to grasp with the receipt as observation (if API allows duck typing or if it's an Observation).
        # API says grasp.observation: None or Observation. EventReceipt is not Observation.
        # So we grasp without observation unless we inspect first.
        # For item 0: No inspection mentioned before pickup.
        await robot.grasp("RIGHT", item_id)
        
        # 4. Move to target (carried move with receipt)
        await robot.move("RIGHT", target_pose, receipt=ready_receipt)
        
        # 5. Clear ready event
        robot.clear_event(ready_event_id, expected_version=ready_receipt.version)
        
        # 6. Release at target
        await robot.release("RIGHT", item_id, target_pose)
        
        # 7. Depart target
        await robot.move("RIGHT", "right_home")
        await robot.release_resource("RIGHT", "buffer_lock")
        
        # 8. Signal empty_0
        robot.signal("empty_0", item_id=item_id)

    # --- Execution ---
    
    # Episode 1: part_0
    # Producer runs, Consumer runs concurrently.
    # Producer: source_0 -> buffer_0 -> left_home. Signal ready_0.
    # Consumer: wait ready_0 -> buffer_0 -> target_0 -> right_home. Signal empty_0.
    t1_producer = asyncio.create_task(producer("part_0", "source_0", "ready_0", "tool"))
    t1_consumer = asyncio.create_task(consumer("part_0", "target_0", "ready_0"))
    await t1_producer
    await t1_consumer

    # Episode 2: part_1
    # Producer: wait empty_0 -> source_1 -> buffer_1 -> left_home. Signal ready_1.
    # Consumer: wait ready_1 -> buffer_1 -> target_1 -> right_home. Signal empty_0.
    
    # Wait for empty_0 (signaled by Consumer 1)
    empty_receipt = await robot.wait_event("empty_0", 10.0)
    robot.clear_event("empty_0", expected_version=empty_receipt.version)
    
    # Inspect both readiness facts (C7 checks serially)
    # "For item 1, wait and clear empty_0, then inspect both readiness facts"
    # Facts: line_clear, receiver_ready
    await robot.inspect("LEFT", "line_clear")
    await robot.inspect("LEFT", "receiver_ready")
    
    # Start Producer and Consumer for part_1
    # Producer uses source_1, ready_1.
    # Consumer uses target_1, ready_1.
    # Note: Consumer 2 also signals empty_0 at the end.
    t2_producer = asyncio.create_task(producer("part_1", "source_1", "ready_1", "tool"))
    t2_consumer = asyncio.create_task(consumer("part_1", "target_1", "ready_1"))
    await t2_producer
    await t2_consumer
    
    # Final state: empty_0 is active (signaled by Consumer 2).
    # Goal: "empty_0 ... inactive at return".
    # Wait and clear empty_0.
    final_empty = await robot.wait_event("empty_0", 10.0)
    robot.clear_event("empty_0", expected_version=final_empty.version)
