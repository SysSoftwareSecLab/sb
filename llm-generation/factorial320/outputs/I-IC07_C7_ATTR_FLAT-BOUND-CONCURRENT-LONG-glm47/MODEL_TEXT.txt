```python
import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot):
    # Constants
    LEFT = "LEFT"
    RIGHT = "RIGHT"
    
    # Poses
    LEFT_HOME = "left_home"
    LEFT_WAIT = "left_wait"
    RIGHT_HOME = "right_home"
    RIGHT_WAIT = "right_wait"
    
    SOURCE_0 = "source_0"
    SOURCE_1 = "source_1"
    BUFFER_0 = "buffer_0"
    BUFFER_1 = "buffer_1"
    TARGET_0 = "target_0"
    TARGET_1 = "target_1"
    
    # Items
    PART_0 = "part_0"
    PART_1 = "part_1"
    
    # Resources
    TOOL = "tool"
    BUFFER_LOCK = "buffer_lock"
    RQ2_GAP_0 = "rq2_gap_0"
    RQ2_GAP_1 = "rq2_gap_1"
    RQ2_GAP_2 = "rq2_gap_2"
    
    # Events
    READY_0 = "ready_0"
    READY_1 = "ready_1"
    EMPTY_0 = "empty_0"
    RQ2_GATE = "rq2_gate"
    
    # Facts
    FACT_LINE_CLEAR = "line_clear"
    FACT_RECEIVER_READY = "receiver_ready"
    
    # Timeouts
    TIMEOUT_S = 4.0
    
    # --- RQ2 Setup ---
    # Signal rq2_gate exactly once
    gate_receipt = robot.signal(RQ2_GATE)
    
    # Acquire and release rq2_gap_0, rq2_gap_1, rq2_gap_2 once each with LEFT, in numeric order
    await robot.acquire(LEFT, RQ2_GAP_0, TIMEOUT_S)
    await robot.release_resource(LEFT, RQ2_GAP_0)
    
    await robot.acquire(LEFT, RQ2_GAP_1, TIMEOUT_S)
    await robot.release_resource(LEFT, RQ2_GAP_1)
    
    await robot.acquire(LEFT, RQ2_GAP_2, TIMEOUT_S)
    await robot.release_resource(LEFT, RQ2_GAP_2)
    
    # Wait for rq2_gate exactly once (after checks)
    await robot.wait_event(RQ2_GATE, TIMEOUT_S)
    
    # --- Inherited Dual-Arm Mission ---
    
    async def producer_part_0():
        # 1. Acquire tool
        await robot.acquire(LEFT, TOOL, TIMEOUT_S)
        
        # 2. Approach source_0 and grasp part_0
        await robot.move(LEFT, SOURCE_0, TIMEOUT_S)
        obs_p0 = await robot.grasp(LEFT, PART_0)
        
        # 3. Acquire buffer_lock
        await robot.acquire(LEFT, BUFFER_LOCK, TIMEOUT_S)
        
        # 4. Move to buffer_0 and release part_0
        await robot.move(LEFT, BUFFER_0, TIMEOUT_S)
        await robot.release(LEFT, PART_0, BUFFER_0)
        
        # 5. Depart buffer (immediate separating departure)
        await robot.move(LEFT, LEFT_HOME, TIMEOUT_S)
        
        # 6. Release buffer_lock
        await robot.release_resource(LEFT, BUFFER_LOCK)
        
        # 7. Publish ready_0
        ready_0_receipt = robot.signal(READY_0)
        
        # 8. Release tool
        await robot.release_resource(LEFT, TOOL)
        
        return ready_0_receipt

    async def consumer_part_0(ready_receipt):
        # 1. Wait for ready_0
        active_ready = await robot.wait_event(READY_0, TIMEOUT_S)
        
        # 2. Acquire buffer_lock
        await robot.acquire(RIGHT, BUFFER_LOCK, TIMEOUT_S)
        
        # 3. Approach buffer_0 and grasp part_0
        await robot.move(RIGHT, BUFFER_0, TIMEOUT_S)
        obs_p0 = await robot.grasp(RIGHT, PART_0)
        
        # 4. Depart buffer
        await robot.move(RIGHT, RIGHT_HOME, TIMEOUT_S)
        
        # 5. Release buffer_lock
        await robot.release_resource(RIGHT, BUFFER_LOCK)
        
        # 6. Move to target_0 carrying the receipt
        await robot.move(RIGHT, TARGET_0, TIMEOUT_S, receipt=active_ready)
        
        # 7. Clear ready_0
        robot.clear_event(READY_0, expected_version=active_ready.version)
        
        # 8. Release part_0
        await robot.release(RIGHT, PART_0, TARGET_0)
        
        # 9. Depart target
        await robot.move(RIGHT, RIGHT_HOME, TIMEOUT_S)
        
        # 10. Publish empty_0
        robot.signal(EMPTY_0)

    async def producer_part_1():
        # 1. Wait and clear empty_0
        empty_receipt = await robot.wait_event(EMPTY_0, TIMEOUT_S)
        robot.clear_event(EMPTY_0, expected_version=empty_receipt.version)
        
        # 2. Inspect both readiness facts (C7 checks serially)
        await robot.inspect(LEFT, FACT_LINE_CLEAR)
        await robot.inspect(LEFT, FACT_RECEIVER_READY)
        
        # 3. Acquire tool
        await robot.acquire(LEFT, TOOL, TIMEOUT_S)
        
        # 4. Approach source_1 and grasp part_1
        await robot.move(LEFT, SOURCE_1, TIMEOUT_S)
        obs_p1 = await robot.grasp(LEFT, PART_1)
        
        # 5. Acquire buffer_lock
        await robot.acquire(LEFT, BUFFER_LOCK, TIMEOUT_S)
        
        # 6. Move to buffer_1 and release part_1
        await robot.move(LEFT, BUFFER_1, TIMEOUT_S)
        await robot.release(LEFT, PART_1, BUFFER_1)
        
        # 7. Depart buffer
        await robot.move(LEFT, LEFT_HOME, TIMEOUT_S)
        
        # 8. Release buffer_lock
        await robot.release_resource(LEFT, BUFFER_LOCK)
        
        # 9. Publish ready_1
        ready_1_receipt = robot.signal(READY_1)
        
        # 10. Release tool
        await robot.release_resource(LEFT, TOOL)

    async def consumer_part_1():
        # 1. Wait for ready_1
        active_ready = await robot.wait_event(READY_1, TIMEOUT_S)
        
        # 2. Acquire buffer_lock
        await robot.acquire(RIGHT, BUFFER_LOCK, TIMEOUT_S)
        
        # 3. Approach buffer_1 and grasp part_1
        await robot.move(RIGHT, BUFFER_1, TIMEOUT_S)
        obs_p1 = await robot.grasp(RIGHT, PART_1)
        
        # 4. Depart buffer
        await robot.move(RIGHT, RIGHT_HOME, TIMEOUT_S)
        
        # 5. Release buffer_lock
        await robot.release_resource(RIGHT, BUFFER_LOCK)
        
        # 6. Move to target_1 carrying the receipt
        await robot.move(RIGHT, TARGET_1, TIMEOUT_S, receipt=active_ready)
        
        # 7. Clear ready_1
        robot.clear_event(READY_1, expected_version=active_ready.version)
        
        # 8. Release part_1
        await robot.release(RIGHT, PART_1, TARGET_1)
        
        # 9. Depart target
        await robot.move(RIGHT, RIGHT_HOME, TIMEOUT_S)

    # Schedule: A alternates, B runs together.
    # We implement B (CONCURRENT) as it is the more complex requirement and satisfies the structure.
    
    # Episode 1: Part 0
    t_prod_0 = asyncio.create_task(producer_part_0())
    t_cons_0 = asyncio.create_task(consumer_part_0(None))
    await t_prod_0
    await t_cons_0
    
    # Episode 2: Part 1
    t_prod_1 = asyncio.create_task(producer_part_1())
    t_cons_1 = asyncio.create_task(consumer_part_1())
    await t_prod_1
    await t_cons_1
    
    # --- Cleanup ---
    # Clear rq2_gate exactly that version after the mission
    robot.clear_event(RQ2_GATE, expected_version=gate_receipt.version)
```
