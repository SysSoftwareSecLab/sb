```python
import asyncio
from bridge_robot_api import Robot


async def run_task(robot: Robot) -> None:
    # ---- rq2_gate protocol ----
    gate_receipt = robot.signal("rq2_gate")

    async def gap_check(rid: str) -> None:
        await robot.acquire("LEFT", rid, 5)
        await robot.release_resource("LEFT", rid)

    await asyncio.gather(
        gap_check("rq2_gap_0"),
        gap_check("rq2_gap_1"),
        gap_check("rq2_gap_2"),
    )

    waited = await robot.wait_event("rq2_gate", 5)
    robot.clear_event("rq2_gate", expected_version=waited.version)

    # ---- inherited dual-arm buffer mission (variant B: concurrent) ----
    empty_receipt_holder: dict[str, object] = {}

    async def producer(part: str, source: str, ready: str, start: str) -> None:
        # First part waits for empty_0; second part signals it after consumer clears.
        if part == "part_1":
            er = await robot.wait_event("empty_0", 50)
            robot.clear_event("empty_0", expected_version=er.version)

        await robot.acquire("LEFT", "buffer_lock", 5)
        await robot.move("LEFT", start)
        await robot.move("LEFT", source)
        await robot.grasp("LEFT", part)
        await robot.move("LEFT", "left_home")
        await robot.move("LEFT", "buffer_0")
        await robot.release("LEFT", part, "buffer_0")
        await robot.move("LEFT", "left_home")
        await robot.release_resource("LEFT", "buffer_lock")
        robot.signal(ready, part)

    async def consumer(part: str, ready: str, target: str) -> None:
        rr = await robot.wait_event(ready, 50)
        await robot.acquire("RIGHT", "buffer_lock", 5)
        await robot.move("RIGHT", "right_home")
        await robot.move("RIGHT", "buffer_0")
        await robot.grasp("RIGHT", part, observation=rr)
        await robot.move("RIGHT", "right_home")
        await robot.move("RIGHT", target, receipt=rr)
        robot.clear_event(ready, expected_version=rr.version)
        await robot.release("RIGHT", part, target)
        await robot.move("RIGHT", "right_home")
        await robot.release_resource("RIGHT", "buffer_lock")
        if part == "part_0":
            er = robot.signal("empty_0", part)
            empty_receipt_holder["receipt"] = er

    await asyncio.gather(
        producer("part_0", "source_0", "ready_0", "left_home"),
        consumer("part_0", "ready_0", "target_0"),
        producer("part_1", "source_1", "ready_1", "left_wait"),
        consumer("part_1", "ready_1", "target_1"),
    )
```
