import asyncio
from bridge_robot_api import Robot, Observation, EventReceipt, ActionReceipt

async def run_task(robot: Robot):
    # --- RQ2 Gate Protocol ---
    # 1. Signal rq2_gate
    gate_receipt = robot.signal("rq2_gate")
    
    # 2. Acquire and release rq2_gap resources in numeric order with LEFT
    #    "Acquire and release rq2_gap_0, rq2_gap_1 and rq2_gap_2 once each with LEFT, in numeric order"
    #    "never retain them at return"
    #    "complete all three rq2_gap resource checks, and only then wait for rq2_gate"
    
    # Gap 0
    await robot.acquire("LEFT", "rq2_gap_0", 4.0)
    await robot.release_resource("LEFT", "rq2_gap_0")
    
    # Gap 1
    await robot.acquire("LEFT", "rq2_gap_1", 4.0)
    await robot.release_resource("LEFT", "rq2_gap_1")
    
    # Gap 2
    await robot.acquire("LEFT", "rq2_gap_2", 4.0)
    await robot.release_resource("LEFT", "rq2_gap_2")
    
    # 3. Wait for rq2_gate (exact active receipt)
    #    "wait its exact active receipt exactly once"
    await robot.wait_event("rq2_gate", 4.0)
    
    # 4. Clear rq2_gate
    #    "clear exactly that version after its assigned protected scope"
    robot.clear_event("rq2_gate", expected_version=gate_receipt.version)

    # --- Inherited Dual-Arm Mission (Producer/Consumer) ---
    #    "A alternates complete producer/consumer episodes"
    #    "LEFT owns tool from before each source pickup through ready publication and releases it on every exit"
    
    # --- Episode 1: part_0 ---
    
    # Producer (LEFT)
    async def producer_part_0():
        # Acquire tool
        await robot.acquire("LEFT", "tool", 4.0)
        
        # Move to source_0 (Approach)
        await robot.move("LEFT", "source_0")
        
        # Grasp part_0
        await robot.grasp("LEFT", "part_0")
        
        # Move to buffer_0
        await robot.move("LEFT", "buffer_0")
        
        # Release part_0 at buffer_0
        await robot.release("LEFT", "part_0", "buffer_0")
        
        # Depart buffer (immediate separating departure)
        await robot.move("LEFT", "left_home")
        
        # Publish ready_0
        robot.signal("ready_0")
        
        # Release tool
        await robot.release_resource("LEFT", "tool")

    # Consumer (RIGHT)
    async def consumer_part_0():
        # Wait for ready_0
        ready_0_receipt = await robot.wait_event("ready_0", 4.0)
        
        # Move to buffer_0 (Approach)
        await robot.move("RIGHT", "buffer_0")
        
        # Grasp part_0
        await robot.grasp("RIGHT", "part_0")
        
        # Move to target_0 with receipt
        await robot.move("RIGHT", "target_0", receipt=ready_0_receipt)
        
        # Clear ready_0 after carried move
        robot.clear_event("ready_0", expected_version=ready_0_receipt.version)
        
        # Release part_0 at target_0
        await robot.release("RIGHT", "part_0", "target_0")
        
        # Depart target
        await robot.move("RIGHT", "right_home")
        
        # Publish empty_0
        robot.signal("empty_0")

    # Run Episode 1 concurrently
    await asyncio.gather(producer_part_0(), consumer_part_0())

    # --- Episode 2: part_1 ---
    
    # Producer (LEFT)
    async def producer_part_1():
        # Wait and clear empty_0 before entering buffer
        empty_0_receipt = await robot.wait_event("empty_0", 4.0)
        robot.clear_event("empty_0", expected_version=empty_0_receipt.version)
        
        # Acquire tool
        await robot.acquire("LEFT", "tool", 4.0)
        
        # Move to source_1 (Approach)
        await robot.move("LEFT", "source_1")
        
        # Grasp part_1
        await robot.grasp("LEFT", "part_1")
        
        # Move to buffer_1
        await robot.move("LEFT", "buffer_1")
        
        # Release part_1 at buffer_1
        await robot.release("LEFT", "part_1", "buffer_1")
        
        # Depart buffer
        await robot.move("LEFT", "left_home")
        
        # Publish ready_1
        robot.signal("ready_1")
        
        # Release tool
        await robot.release_resource("LEFT", "tool")

    # Consumer (RIGHT)
    async def consumer_part_1():
        # Wait for ready_1
        ready_1_receipt = await robot.wait_event("ready_1", 4.0)
        
        # Move to buffer_1 (Approach)
        await robot.move("RIGHT", "buffer_1")
        
        # Grasp part_1
        await robot.grasp("RIGHT", "part_1")
        
        # Move to target_1 with receipt
        await robot.move("RIGHT", "target_1", receipt=ready_1_receipt)
        
        # Clear ready_1 after carried move
        robot.clear_event("ready_1", expected_version=ready_1_receipt.version)
        
        # Release part_1 at target_1
        await robot.release("RIGHT", "part_1", "target_1")
        
        # Depart target
        await robot.move("RIGHT", "right_home")

    # Run Episode 2 concurrently
    await asyncio.gather(producer_part_1(), consumer_part_1())
