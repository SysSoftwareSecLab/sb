import asyncio
from bridge_robot_api import MotionFault

async def run_task(robot):
    # ============================================================
    # IC08_C8_ATTR_NESTED-BOUND-CONCURRENT-LONG
    # LEFT = producer (A), RIGHT = consumer (B)
    # ============================================================

    # ---- rq2_gate lifecycle: signal -> gap checks -> wait -> mission -> clear
    gate_receipt = robot.signal("rq2_gate")

    # Acquire rq2_gap_0, rq2_gap_1, rq2_gap_2 once each with LEFT, numeric order
    for gap in ("rq2_gap_0", "rq2_gap_1", "rq2_gap_2"):
        await robot.acquire("LEFT", gap, 5)
        await robot.release_resource("LEFT", gap)

    # Wait for the exact active rq2_gate receipt
    gate_wait = await robot.wait_event("rq2_gate", 5)

    # ---- Inherited dual-arm buffer mission (concurrent with gate) ----
    mission = asyncio.create_task(_dual_arm_mission(robot))
    await mission

    # Clear rq2_gate after the protected scope
    robot.clear_event("rq2_gate", expected_version=gate_receipt.version)


async def _dual_arm_mission(robot):
    """A alternates complete producer/consumer episodes; B runs producer and
    consumer coroutines together."""
    await asyncio.gather(
        _left_episode(robot, "part_0", "source_0", "buffer_0",
                      "ready_0", "target_0", "left_home"),
        _right_episode(robot, "part_0", "buffer_0", "ready_0",
                       "target_0", "right_home"),
    )
    await asyncio.gather(
        _left_episode(robot, "part_1", "source_1", "buffer_1",
                      "ready_1", "target_1", "left_wait"),
        _right_episode(robot, "part_1", "buffer_1", "ready_1",
                       "target_1", "right_wait"),
    )


async def _left_episode(robot, part, source, buffer, ready_event,
                        target, return_pose):
    """LEFT producer: source -> buffer, publish ready, depart, wait empty."""
    # LEFT owns tool from before each source pickup through ready publication
    await robot.acquire("LEFT", "tool", 5)

    # Approach source (start_pose -> interaction_pose), immediately grasp
    await robot.move("LEFT", source, 4)
    await robot.grasp("LEFT", part)

    # Both participants own buffer_lock during buffer entry and departure
    await robot.acquire("LEFT", "buffer_lock", 5)
    await robot.move("LEFT", buffer, 4)

    # Release part at buffer, immediately depart
    await robot.release("LEFT", part, buffer)
    await robot.move("LEFT", return_pose, 4)
    await robot.release_resource("LEFT", "buffer_lock")

    # Publish ready after departing buffer
    ready_receipt = robot.signal(ready_event, part)

    # Wait for empty_0 before entering buffer with the second part
    empty_receipt = await robot.wait_event("empty_0", 30)

    # For item 1: inspect both readiness facts (serial C7-style check)
    if part == "part_1":
        await robot.inspect("LEFT", "line_clear")
        await robot.inspect("LEFT", "receiver_ready")

    # Release tool on every exit
    await robot.release_resource("LEFT", "tool")

    # Clear empty_0 (producer waits and clears empty_0 before second entry)
    robot.clear_event("empty_0", expected_version=empty_receipt.version)


async def _right_episode(robot, part, buffer, ready_event, target, return_pose):
    """RIGHT consumer: wait ready, pickup from buffer, carry to target,
    clear ready, release, depart, publish empty_0."""
    # Wait the corresponding ready receipt before pickup
    ready_receipt = await robot.wait_event(ready_event, 30)

    # Both participants own buffer_lock during buffer entry and departure
    await robot.acquire("RIGHT", "buffer_lock", 5)

    # Approach buffer (start_pose -> interaction_pose), immediately grasp
    await robot.move("RIGHT", buffer, 4)
    await robot.grasp("RIGHT", part, observation=ready_receipt)

    # Supply that exact active item receipt on carried move to target
    await robot.move("RIGHT", target, 4, receipt=ready_receipt)

    # Consumer clears ready after its carried move
    robot.clear_event(ready_event, expected_version=ready_receipt.version)

    # Release on target and depart before publishing empty_0
    await robot.release("RIGHT", part, target)
    await robot.move("RIGHT", return_pose, 4)
    await robot.release_resource("RIGHT", "buffer_lock")

    # Publish empty_0
    robot.signal("empty_0", part)
